import React from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';

// Context Providers
import { AuthProvider } from './context/AuthContext';
import { BookingProvider } from './context/BookingContext';

// Components
import Header from './components/Header';
import Footer from './components/Footer';
import ProtectedRoute from './components/ProtectedRoute';

// Pages
import HomePage from './pages/HomePage';
import LoginPage from './pages/LoginPage';
import MoviesPage from './pages/MoviesPage';
import BookingPage from './pages/BookingPage';
import SummaryPage from './pages/SummaryPage';
import NotFound from './pages/NotFound';

import './App.css';

function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <BookingProvider>
          <div className="app-wrapper">
            <Header />
            <main className="main-content">
              <Routes>
                {/* Public Routes */}
                <Route path="/" element={<HomePage />} />
                <Route path="/login" element={<LoginPage />} />

                {/* Protected Routes (Authentication Required) */}
                <Route
                  path="/movies"
                  element={
                    <ProtectedRoute>
                      <MoviesPage />
                    </ProtectedRoute>
                  }
                />
                <Route
                  path="/booking"
                  element={
                    <ProtectedRoute>
                      <BookingPage />
                    </ProtectedRoute>
                  }
                />
                <Route
                  path="/summary"
                  element={
                    <ProtectedRoute>
                      <SummaryPage />
                    </ProtectedRoute>
                  }
                />

                {/* 404 Route */}
                <Route path="*" element={<NotFound />} />
              </Routes>
            </main>
            <Footer />
          </div>
        </BookingProvider>
      </AuthProvider>
    </BrowserRouter>
  );
}

export default App;

import React, { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import useAuth from '../hooks/useAuth';

function LoginPage() {
  const [usernameInput, setUsernameInput] = useState('');
  const [passwordInput, setPasswordInput] = useState('');
  const [errorMessage, setErrorMessage] = useState('');

  const { login, isAuthenticated } = useAuth();
  const navigate = useNavigate();

  // If already logged in, redirect immediately to /movies
  useEffect(() => {
    if (isAuthenticated) {
      navigate('/movies', { replace: true });
    }
  }, [isAuthenticated, navigate]);

  const handleSubmit = (e) => {
    e.preventDefault();
    setErrorMessage('');

    if (!usernameInput.trim() || !passwordInput.trim()) {
      setErrorMessage('Please enter both username and password.');
      return;
    }

    const result = login(usernameInput.trim(), passwordInput.trim());
    if (result.success) {
      navigate('/movies');
    } else {
      setErrorMessage(result.message);
    }
  };

  return (
    <div className="login-wrapper">
      <div className="login-card">
        {/* Cinema Brand Header */}
        <div className="login-header">
          <div className="login-icon">🎬</div>
          <h2 className="login-title">BookMyMovie</h2>
          <p className="login-subtitle">Sign in to book your movie tickets</p>
        </div>

        {/* Error message */}
        {errorMessage && (
          <div className="error-alert">
            <span>⚠️</span>
            <span>{errorMessage}</span>
          </div>
        )}

        {/* Form */}
        <form onSubmit={handleSubmit} className="login-form">
          <div className="form-group">
            <label htmlFor="username">Username</label>
            <input
              id="username"
              type="text"
              className="form-input"
              placeholder="Enter username"
              value={usernameInput}
              onChange={(e) => setUsernameInput(e.target.value)}
              autoComplete="username"
            />
          </div>

          <div className="form-group">
            <label htmlFor="password">Password</label>
            <input
              id="password"
              type="password"
              className="form-input"
              placeholder="Enter password"
              value={passwordInput}
              onChange={(e) => setPasswordInput(e.target.value)}
              autoComplete="current-password"
            />
          </div>

          <button type="submit" className="btn-primary login-btn">
            Sign In
          </button>
        </form>

        {/* Demo Credentials Box */}
        <div className="demo-credentials-box">
          <div className="demo-badge">🔑 Demo Credentials</div>
          <div className="demo-details">
            <p><strong>Username:</strong> <code>admin</code></p>
            <p><strong>Password:</strong> <code>1234</code></p>
          </div>
        </div>
      </div>
    </div>
  );
}

export default LoginPage;

import React from 'react';
import { Navigate } from 'react-router-dom';
import useAuth from '../hooks/useAuth';

// ProtectedRoute component redirects unauthenticated users to /login
function ProtectedRoute({ children }) {
  const { isAuthenticated } = useAuth();

  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  return children;
}

export default ProtectedRoute;

import React from 'react';
import useBooking from '../hooks/useBooking';

// Poster icons for the 4 lab movies
const POSTER_ICONS = {
  1: '🦁', // Leo
  2: '🐐', // GOAT
  3: '🐉', // Dragon
  4: '⚡', // Coolie
};

function MovieList({ onSelectAndContinue }) {
  const { filteredMovies, selectedMovie, handleSelectMovie } = useBooking();

  const handleCardClick = (movie) => {
    handleSelectMovie(movie);
    if (onSelectAndContinue) {
      onSelectAndContinue(movie);
    }
  };

  return (
    <div>
      {filteredMovies.length === 0 ? (
        <div style={{ textAlign: 'center', padding: '40px 20px', color: '#94a3b8' }}>
          <p style={{ fontSize: '18px' }}>No movies found matching your search.</p>
        </div>
      ) : (
        <div className="movies-grid">
          {filteredMovies.map((movie) => {
            const isSelected = selectedMovie?.id === movie.id;
            return (
              <div
                key={movie.id}
                onClick={() => handleCardClick(movie)}
                className={`movie-card ${isSelected ? 'selected' : ''}`}
              >
                <div className="movie-poster-mock">
                  {POSTER_ICONS[movie.id] || '🎬'}
                </div>

                <div className="movie-card-info">
                  <h4>{movie.title}</h4>
                  <div className="movie-meta">
                    <span>{movie.genre}</span>
                    <span>{movie.year}</span>
                  </div>
                </div>

                <button
                  type="button"
                  className="movie-select-btn"
                  onClick={(e) => {
                    e.stopPropagation();
                    handleCardClick(movie);
                  }}
                >
                  {isSelected ? '✓ Selected' : 'Select Movie'}
                </button>
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}

export default MovieList;

import React, { createContext, useState, useEffect } from 'react';

// Create AuthContext
const AuthContext = createContext();

const DEMO_USER = 'admin';
const DEMO_PASS = '1234';

function AuthProvider({ children }) {
  // Initialize auth state from localStorage to persist login across page refreshes
  const [username, setUsername] = useState(() => {
    return localStorage.getItem('auth_user') || '';
  });

  const [isAuthenticated, setIsAuthenticated] = useState(() => {
    return localStorage.getItem('auth_status') === 'true';
  });

  // Login function
  const login = (inputUser, inputPass) => {
    if (inputUser === DEMO_USER && inputPass === DEMO_PASS) {
      setUsername(inputUser);
      setIsAuthenticated(true);
      localStorage.setItem('auth_user', inputUser);
      localStorage.setItem('auth_status', 'true');
      return { success: true };
    } else {
      return { success: false, message: 'Invalid username or password.' };
    }
  };

  // Logout function
  const logout = () => {
    setUsername('');
    setIsAuthenticated(false);
    localStorage.removeItem('auth_user');
    localStorage.removeItem('auth_status');
  };

  const value = {
    username,
    isAuthenticated,
    login,
    logout,
  };

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

export { AuthContext, AuthProvider };

import React, { createContext, useState, useMemo, useCallback } from 'react';

// Create Context
const BookingContext = createContext();

// Movie data as per Lab Manual (Experiment 1, 2, 3)
const MOVIES = [
  { id: 1, title: 'Leo',    genre: 'Action',  year: 2023 },
  { id: 2, title: 'GOAT',   genre: 'Action',  year: 2024 },
  { id: 3, title: 'Dragon', genre: 'Fantasy', year: 2025 },
  { id: 4, title: 'Coolie', genre: 'Action',  year: 2025 },
];

const TICKET_PRICE = 150;

function BookingProvider({ children }) {
  const [searchQuery, setSearchQuery]           = useState('');
  const [selectedMovie, setSelectedMovie]       = useState(null);
  const [selectedTheatre, setSelectedTheatre]   = useState(null);
  const [selectedTime, setSelectedTime]         = useState(null);
  const [selectedSeats, setSelectedSeats]       = useState([]);
  const [bookingConfirmed, setBookingConfirmed] = useState(false);

  // Filter movies based on search query
  const filteredMovies = useMemo(() => {
    if (!searchQuery.trim()) return MOVIES;
    return MOVIES.filter((movie) =>
      movie.title.toLowerCase().includes(searchQuery.toLowerCase())
    );
  }, [searchQuery]);

  // Total amount calculation
  const totalAmount = useMemo(() => {
    return selectedSeats.length * TICKET_PRICE;
  }, [selectedSeats]);

  // Select movie handler
  const handleSelectMovie = useCallback((movie) => {
    setSelectedMovie(movie);
    setSelectedTheatre(null);
    setSelectedTime(null);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  // Select theatre handler
  const handleSelectTheatre = useCallback((theatre) => {
    setSelectedTheatre(theatre);
    setSelectedTime(null);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  // Select show time handler
  const handleSelectTime = useCallback((timing) => {
    setSelectedTime(timing);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  // Toggle seat selection
  const handleToggleSeat = useCallback((seat) => {
    setSelectedSeats((prevSeats) => {
      if (prevSeats.includes(seat)) {
        return prevSeats.filter((s) => s !== seat);
      } else {
        return [...prevSeats, seat];
      }
    });
    setBookingConfirmed(false);
  }, []);

  // Search input handler
  const handleSearchChange = useCallback((value) => {
    setSearchQuery(value);
  }, []);

  // Confirm booking handler
  const handleConfirmBooking = useCallback(() => {
    setBookingConfirmed(true);
  }, []);

  // Reset booking handler
  const handleResetBooking = useCallback(() => {
    setSelectedMovie(null);
    setSelectedTheatre(null);
    setSelectedTime(null);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  const contextValue = {
    // State
    searchQuery,
    selectedMovie,
    selectedTheatre,
    selectedTime,
    selectedSeats,
    bookingConfirmed,
    filteredMovies,
    totalAmount,
    ticketPrice: TICKET_PRICE,
    // Handlers
    handleSearchChange,
    handleSelectMovie,
    handleSelectTheatre,
    handleSelectTime,
    handleToggleSeat,
    handleConfirmBooking,
    handleResetBooking,
  };

  return (
    <BookingContext.Provider value={contextValue}>
      {children}
    </BookingContext.Provider>
  );
}

export { BookingContext, BookingProvider };
