import { createSlice } from '@reduxjs/toolkit';
import { calculateSeatsPriceBreakdown, SEAT_PRICES } from '../data/seatLayout.js';
import { getAvailableBookingDates } from '../data/showtimes.js';

const defaultDates = getAvailableBookingDates(7);

const initialPricing = {
  Regular: { count: 0, rate: SEAT_PRICES?.Regular || 150, total: 0, seats: [] },
  Premium: { count: 0, rate: SEAT_PRICES?.Premium || 200, total: 0, seats: [] },
  Recliner: { count: 0, rate: SEAT_PRICES?.Recliner || 300, total: 0, seats: [] },
  totalAmount: 0,
  totalSeats: 0,
};

const initialState = {
  selectedMovie: null,
  selectedTheatre: null,
  selectedDate: defaultDates[0] || null,
  selectedTime: null,
  selectedSeats: [],
  totalAmount: 0,
  totalSeats: 0,
  priceBreakdown: initialPricing,
  bookingConfirmed: false,
  bookingReference: null,
  bookingError: null,
  searchQuery: '',
  theatreAreaFilter: 'All Areas',
  theatreSearchQuery: '',
};

export const bookingSlice = createSlice({
  name: 'booking',
  initialState,
  reducers: {
    // 1. Action to select a movie and reset downstream selections
    setMovie: (state, action) => {
      state.selectedMovie = action.payload;
      state.selectedTheatre = null;
      state.selectedDate = defaultDates[0] || null;
      state.selectedTime = null;
      state.selectedSeats = [];
      state.totalAmount = 0;
      state.totalSeats = 0;
      state.priceBreakdown = initialPricing;
      state.bookingConfirmed = false;
      state.bookingReference = null;
      state.bookingError = null;
    },

    // 2. Action to select a cinema theatre
    setTheatre: (state, action) => {
      state.selectedTheatre = action.payload;
      state.selectedTime = null;
      state.selectedSeats = [];
      state.totalAmount = 0;
      state.totalSeats = 0;
      state.priceBreakdown = initialPricing;
      state.bookingConfirmed = false;
      state.bookingReference = null;
      state.bookingError = null;
    },

    // 3. Action to select booking calendar date
    setShowDate: (state, action) => {
      state.selectedDate = action.payload;
      state.selectedTime = null;
      state.selectedSeats = [];
      state.totalAmount = 0;
      state.totalSeats = 0;
      state.priceBreakdown = initialPricing;
      state.bookingConfirmed = false;
      state.bookingReference = null;
      state.bookingError = null;
    },

    // 4. Action to select show timing
    setShowTime: (state, action) => {
      state.selectedTime = action.payload;
      state.selectedSeats = [];
      state.totalAmount = 0;
      state.totalSeats = 0;
      state.priceBreakdown = initialPricing;
      state.bookingConfirmed = false;
      state.bookingReference = null;
      state.bookingError = null;
    },

    // 5. Action to toggle seat selection and automatically recalculate price
    toggleSeat: (state, action) => {
      const seatId = action.payload;
      const index = state.selectedSeats.indexOf(seatId);

      if (index >= 0) {
        state.selectedSeats.splice(index, 1);
      } else {
        state.selectedSeats.push(seatId);
      }

      const breakdown = calculateSeatsPriceBreakdown(state.selectedSeats);
      state.priceBreakdown = breakdown;
      state.totalAmount = breakdown.totalAmount;
      state.totalSeats = breakdown.totalSeats;
      state.bookingConfirmed = false;
      state.bookingError = null;
    },

    // 6. Action to remove a specific seat and update grand total
    removeSeat: (state, action) => {
      const seatId = action.payload;
      state.selectedSeats = state.selectedSeats.filter((s) => s !== seatId);

      const breakdown = calculateSeatsPriceBreakdown(state.selectedSeats);
      state.priceBreakdown = breakdown;
      state.totalAmount = breakdown.totalAmount;
      state.totalSeats = breakdown.totalSeats;
      state.bookingConfirmed = false;
      state.bookingError = null;
    },

    // 7. Action to update search filter
    setSearchQuery: (state, action) => {
      state.searchQuery = action.payload;
    },

    // 8. Actions to filter Coimbatore theatres
    setTheatreAreaFilter: (state, action) => {
      state.theatreAreaFilter = action.payload;
    },

    setTheatreSearchQuery: (state, action) => {
      state.theatreSearchQuery = action.payload;
    },

    // 9. Action on successful booking confirmation
    confirmBookingSuccess: (state, action) => {
      state.bookingConfirmed = true;
      state.bookingReference = action.payload;
      state.bookingError = null;
    },

    // 10. Action to set booking validation error
    setBookingError: (state, action) => {
      state.bookingError = action.payload;
    },

    // 11. Action to reset entire booking state
    clearBooking: (state) => {
      state.selectedMovie = null;
      state.selectedTheatre = null;
      state.selectedDate = defaultDates[0] || null;
      state.selectedTime = null;
      state.selectedSeats = [];
      state.totalAmount = 0;
      state.totalSeats = 0;
      state.priceBreakdown = initialPricing;
      state.bookingConfirmed = false;
      state.bookingReference = null;
      state.bookingError = null;
      state.searchQuery = '';
      state.theatreAreaFilter = 'All Areas';
      state.theatreSearchQuery = '';
    },
  },
});

export const {
  setMovie,
  setTheatre,
  setShowDate,
  setShowTime,
  toggleSeat,
  removeSeat,
  setSearchQuery,
  setTheatreAreaFilter,
  setTheatreSearchQuery,
  confirmBookingSuccess,
  setBookingError,
  clearBooking,
} = bookingSlice.actions;

export default bookingSlice.reducer;

/**
 * Custom Redux Middleware for Movie Booking Actions:
 * - Logs action dispatch events with timestamp and state previews.
 * - Validates booking constraints (e.g., maximum 10 seats per transaction).
 * - Synchronizes active booking progress to browser storage for audit and recovery.
 */
export const bookingMiddleware = (store) => (next) => (action) => {
  // 1. Seat limit enforcement rule (Max 10 seats per booking)
  if (action.type === 'booking/toggleSeat') {
    const currentState = store.getState().booking;
    const seatId = action.payload;
    const isAlreadySelected = currentState.selectedSeats.includes(seatId);

    if (!isAlreadySelected && currentState.selectedSeats.length >= 10) {
      alert('Maximum seat limit reached! You can select up to 10 seats per booking.');
      return; // Intercept and block action from reaching the reducer
    }
  }

  // 2. Dispatch action through reducer pipeline
  const result = next(action);

  // 3. Action audit and logging
  if (typeof action.type === 'string' && action.type.startsWith('booking/')) {
    const nextState = store.getState().booking;
    console.group(`[Redux Booking Middleware] Action: ${action.type}`);
    console.log('Timestamp:', new Date().toLocaleTimeString());
    console.log('Action Payload:', action.payload);
    console.log('Selected Seats:', nextState.selectedSeats);
    console.log('Calculated Grand Total: ₹' + nextState.totalAmount);
    console.groupEnd();
  }

  // 4. Persistence handling for confirmed bookings
  if (action.type === 'booking/confirmBookingSuccess') {
    const finalState = store.getState().booking;
    const passData = {
      bookingReference: action.payload,
      movie: finalState.selectedMovie?.title,
      theatre: finalState.selectedTheatre?.name,
      date: finalState.selectedDate?.displayLabel || finalState.selectedDate,
      time: finalState.selectedTime?.time || finalState.selectedTime,
      seats: finalState.selectedSeats,
      totalAmount: finalState.totalAmount,
      timestamp: new Date().toISOString(),
    };
    try {
      localStorage.setItem('bm_redux_confirmed_pass', JSON.stringify(passData));
    } catch (err) {
      console.warn('Storage sync failed in bookingMiddleware:', err);
    }
  }

  return result;
};

export default bookingMiddleware;

import { useState, useCallback, useMemo } from 'react';
import { useSelector, useDispatch } from 'react-redux';
import {
  setMovie,
  setTheatre,
  setShowDate,
  setShowTime,
  toggleSeat,
  removeSeat,
  setSearchQuery,
  setTheatreAreaFilter as setReduxTheatreAreaFilter,
  setTheatreSearchQuery as setReduxTheatreSearchQuery,
  confirmBookingSuccess,
  setBookingError,
  clearBooking,
} from '../redux/bookingSlice';
import { COIMBATORE_THEATRES } from '../data/theatres.js';
import { getAvailableBookingDates } from '../data/showtimes.js';
import { ROW_CONFIG, SEAT_PRICES, TOTAL_SEATS_PER_SCREEN } from '../data/seatLayout.js';
import {
  buildShowKey,
  getBookedSeatsForShow,
  confirmNewBooking,
  getAllSavedBookings,
} from '../services/bookingService.js';
import { FALLBACK_MOVIES_2026 } from '../data/fallbackMovies.js';

export function useBooking() {
  const dispatch = useDispatch();
  const bookingState = useSelector((state) => state.booking);

  // Available 8 days (Today + 7 days)
  const availableDates = useMemo(() => getAvailableBookingDates(7), []);
  const selectedDate = bookingState.selectedDate || availableDates[0];

  // Saved bookings from localStorage (reactive state)
  const [allBookings, setAllBookings] = useState(() => getAllSavedBookings());

  const lastBooking = useMemo(() => {
    return allBookings && allBookings.length > 0 ? allBookings[0] : null;
  }, [allBookings]);

  // Current screening key based on Redux selection
  const currentShowKey = useMemo(() => {
    if (
      !bookingState.selectedMovie ||
      !bookingState.selectedTheatre ||
      !selectedDate ||
      !bookingState.selectedTime
    ) {
      return null;
    }
    return buildShowKey(
      bookingState.selectedMovie.id,
      bookingState.selectedTheatre.id,
      selectedDate.iso || selectedDate,
      bookingState.selectedTime.id || bookingState.selectedTime.time
    );
  }, [
    bookingState.selectedMovie,
    bookingState.selectedTheatre,
    selectedDate,
    bookingState.selectedTime,
  ]);

  // Booked seats for current show
  const bookedSeatsForCurrentShow = useMemo(() => {
    if (!currentShowKey) return [];
    return getBookedSeatsForShow(currentShowKey, bookingState.bookingConfirmed);
  }, [currentShowKey, bookingState.bookingConfirmed]);

  // Filter movies by search query
  const filteredMovies = useMemo(() => {
    const q = (bookingState.searchQuery || '').trim().toLowerCase();
    if (!q) return FALLBACK_MOVIES_2026;
    return FALLBACK_MOVIES_2026.filter((movie) =>
      movie.title.toLowerCase().includes(q)
    );
  }, [bookingState.searchQuery]);

  // Redux Action Dispatchers
  const handleSelectMovie = useCallback(
    (movie) => {
      dispatch(setMovie(movie));
    },
    [dispatch]
  );

  const handleSelectTheatre = useCallback(
    (theatre) => {
      dispatch(setTheatre(theatre));
    },
    [dispatch]
  );

  const handleSelectDate = useCallback(
    (date) => {
      dispatch(setShowDate(date));
    },
    [dispatch]
  );

  const handleSelectTime = useCallback(
    (timing) => {
      dispatch(setShowTime(timing));
    },
    [dispatch]
  );

  const handleToggleSeat = useCallback(
    (seatId) => {
      if (bookedSeatsForCurrentShow.includes(seatId)) {
        return false;
      }
      dispatch(toggleSeat(seatId));
      return true;
    },
    [dispatch, bookedSeatsForCurrentShow]
  );

  const handleRemoveSeat = useCallback(
    (seatId) => {
      dispatch(removeSeat(seatId));
    },
    [dispatch]
  );

  const handleSearchChange = useCallback(
    (value) => {
      dispatch(setSearchQuery(value));
    },
    [dispatch]
  );

  const setTheatreAreaFilter = useCallback(
    (area) => {
      dispatch(setReduxTheatreAreaFilter(area));
    },
    [dispatch]
  );

  const setTheatreSearchQuery = useCallback(
    (query) => {
      dispatch(setReduxTheatreSearchQuery(query));
    },
    [dispatch]
  );

  const handleConfirmBooking = useCallback(() => {
    if (
      !bookingState.selectedMovie ||
      !bookingState.selectedTheatre ||
      !selectedDate ||
      !bookingState.selectedTime ||
      bookingState.selectedSeats.length === 0
    ) {
      dispatch(
        setBookingError('Please complete all booking steps (Theatre, Date, Showtime, and at least 1 Seat).')
      );
      return false;
    }

    try {
      const record = confirmNewBooking({
        movie: bookingState.selectedMovie,
        theatre: bookingState.selectedTheatre,
        date: selectedDate,
        showtime: bookingState.selectedTime,
        seats: bookingState.selectedSeats,
      });

      dispatch(confirmBookingSuccess(record.bookingId));
      setAllBookings(getAllSavedBookings());
      return true;
    } catch (err) {
      dispatch(
        setBookingError(err.message || 'Unable to confirm booking. Please try again.')
      );
      return false;
    }
  }, [
    dispatch,
    bookingState.selectedMovie,
    bookingState.selectedTheatre,
    selectedDate,
    bookingState.selectedTime,
    bookingState.selectedSeats,
  ]);

  const handleResetBooking = useCallback(() => {
    dispatch(clearBooking());
  }, [dispatch]);

  return {
    // Redux State
    selectedMovie: bookingState.selectedMovie,
    selectedTheatre: bookingState.selectedTheatre,
    selectedDate,
    selectedTime: bookingState.selectedTime,
    selectedSeats: bookingState.selectedSeats,
    totalAmount: bookingState.totalAmount,
    totalSeats: bookingState.totalSeats,
    priceBreakdown: bookingState.priceBreakdown,
    bookingConfirmed: bookingState.bookingConfirmed,
    activeBookingReference: bookingState.bookingReference,
    bookingError: bookingState.bookingError,
    searchQuery: bookingState.searchQuery,
    theatreAreaFilter: bookingState.theatreAreaFilter || 'All Areas',
    theatreSearchQuery: bookingState.theatreSearchQuery || '',

    // Static & Derived
    theatres: COIMBATORE_THEATRES,
    availableDates,
    currentShowKey,
    bookedSeatsForCurrentShow,
    filteredMovies,
    seatPrices: SEAT_PRICES,
    ticketPrice: SEAT_PRICES.Regular,
    rowConfig: ROW_CONFIG,
    totalSeatsPerScreen: TOTAL_SEATS_PER_SCREEN,
    allBookings,
    lastBooking,

    // Dispatch Handlers
    handleSelectMovie,
    handleSelectTheatre,
    handleSelectDate,
    handleSelectTime,
    handleToggleSeat,
    handleRemoveSeat,
    handleSearchChange,
    handleConfirmBooking,
    handleResetBooking,
    setTheatreAreaFilter,
    setTheatreSearchQuery,
  };
}

export default useBooking;

import React from 'react';
import MovieCard from './MovieCard';
import Loading from './Loading';
import ErrorMessage from './ErrorMessage';
import EmptyState from './EmptyState';
import useBooking from '../hooks/useBooking';

function MovieList({
  movies = [],
  isLoading = false,
  error = null,
  onRetry,
  onReset,
  onSelectAndContinue,
}) {
  const { selectedMovie, handleSelectMovie } = useBooking();

  if (isLoading) {
    return <Loading type="skeleton" count={8} />;
  }

  if (error) {
    return (
      <ErrorMessage
        title="Failed to Load Movies"
        message={error}
        onRetry={onRetry}
        retryText="Retry Loading"
      />
    );
  }

  if (!movies || movies.length === 0) {
    return (
      <EmptyState
        title="No Movies Found"
        message="No cinema titles matched your current search query or genre filter. Try adjusting your search."
        onReset={onReset}
      />
    );
  }

  const handleMovieSelect = (movie) => {
    handleSelectMovie(movie);
    if (onSelectAndContinue) {
      onSelectAndContinue(movie);
    }
  };

  return (
    <section className="movie-list-section" aria-label="Movies Collection">
      <div className="movies-grid">
        {movies.map((movie) => {
          const isSelected = selectedMovie && String(selectedMovie.id) === String(movie.id);
          return (
            <MovieCard
              key={movie.id}
              movie={movie}
              isSelected={isSelected}
              onSelect={handleMovieSelect}
            />
          );
        })}
      </div>
    </section>
  );
}

export default MovieList;

import React, { useState, useMemo } from 'react';
import useBooking from '../hooks/useBooking';
import { COIMBATORE_THEATRES, getCoimbatoreAreas } from '../data/theatres.js';

function TheatreList() {
  const { selectedTheatre, handleSelectTheatre } = useBooking();
  const [theatreAreaFilter, setTheatreAreaFilter] = useState('All Areas');
  const [theatreSearchQuery, setTheatreSearchQuery] = useState('');

  const areas = useMemo(() => getCoimbatoreAreas(), []);

  const filteredTheatres = useMemo(() => {
    return COIMBATORE_THEATRES.filter((theatre) => {
      const matchesArea =
        theatreAreaFilter === 'All Areas' ||
        theatre.area.toLowerCase() === theatreAreaFilter.toLowerCase();

      const q = theatreSearchQuery.trim().toLowerCase();
      const matchesSearch =
        !q ||
        theatre.name.toLowerCase().includes(q) ||
        theatre.area.toLowerCase().includes(q) ||
        theatre.address.toLowerCase().includes(q) ||
        theatre.landmark.toLowerCase().includes(q);

      return matchesArea && matchesSearch;
    });
  }, [theatreAreaFilter, theatreSearchQuery]);

  return (
    <div className="theatre-selection-container">
      <div className="theatre-filter-toolbar">
        <div className="theatre-search-box">
          <span className="search-icon">🔍</span>
          <input
            type="text"
            className="theatre-search-input"
            value={theatreSearchQuery}
            onChange={(e) => setTheatreSearchQuery(e.target.value)}
            placeholder="Search Coimbatore theatres by name, area, or road..."
          />
          {theatreSearchQuery && (
            <button
              type="button"
              className="clear-search-btn"
              onClick={() => setTheatreSearchQuery('')}
            >
              ✕
            </button>
          )}
        </div>

        <div className="area-filter-pills">
          {areas.map((area) => (
            <button
              key={area}
              type="button"
              onClick={() => setTheatreAreaFilter(area)}
              className={`area-pill ${theatreAreaFilter === area ? 'active' : ''}`}
            >
              {area}
            </button>
          ))}
        </div>
      </div>

      <div className="theatres-grid">
        {filteredTheatres.map((theatre) => {
          const isSelected = selectedTheatre?.id === theatre.id;
          return (
            <div
              key={theatre.id}
              onClick={() => handleSelectTheatre(theatre)}
              className={`theatre-card ${isSelected ? 'selected' : ''}`}
            >
              <div className="theatre-card-top">
                <span className="theatre-name">{theatre.name}</span>
                <span className="theatre-area-tag">{theatre.area}</span>
              </div>
              <p className="theatre-address">{theatre.address}</p>
              <div className="theatre-meta-footer">
                <span className="theatre-screens">{theatre.screens}</span>
                <button type="button" className={`theatre-select-btn ${isSelected ? 'selected' : ''}`}>
                  {isSelected ? '✓ Selected' : 'Select'}
                </button>
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}

export default TheatreList;

export const SEAT_PRICES = {
  Regular: 150,
  Premium: 200,
  Recliner: 300,
};

export const ROW_CONFIG = [
  { row: 'A', category: 'Regular',  price: SEAT_PRICES.Regular,  seatsCount: 11 },
  { row: 'B', category: 'Regular',  price: SEAT_PRICES.Regular,  seatsCount: 11 },
  { row: 'C', category: 'Regular',  price: SEAT_PRICES.Regular,  seatsCount: 11 },
  { row: 'D', category: 'Regular',  price: SEAT_PRICES.Regular,  seatsCount: 11 },
  { row: 'E', category: 'Regular',  price: SEAT_PRICES.Regular,  seatsCount: 11 },
  { row: 'F', category: 'Premium',  price: SEAT_PRICES.Premium,  seatsCount: 11 },
  { row: 'G', category: 'Premium',  price: SEAT_PRICES.Premium,  seatsCount: 11 },
  { row: 'H', category: 'Premium',  price: SEAT_PRICES.Premium,  seatsCount: 11 },
  { row: 'I', category: 'Recliner', price: SEAT_PRICES.Recliner, seatsCount: 11 },
  { row: 'J', category: 'Recliner', price: SEAT_PRICES.Recliner, seatsCount: 11 },
];

export const TOTAL_SEATS_PER_SCREEN = 110;

export function getSeatCategory(seatId) {
  if (!seatId) return 'Regular';
  const row = String(seatId).trim().charAt(0).toUpperCase();
  const found = ROW_CONFIG.find((r) => r.row === row);
  return found ? found.category : 'Regular';
}

export function getSeatPrice(seatId) {
  if (!seatId) return SEAT_PRICES.Regular;
  const category = getSeatCategory(seatId);
  return SEAT_PRICES[category] || SEAT_PRICES.Regular;
}

export function calculateSeatsPriceBreakdown(seatIds = []) {
  const breakdown = {
    Regular:  { count: 0, rate: SEAT_PRICES.Regular,  total: 0, seats: [] },
    Premium:  { count: 0, rate: SEAT_PRICES.Premium,  total: 0, seats: [] },
    Recliner: { count: 0, rate: SEAT_PRICES.Recliner, total: 0, seats: [] },
    totalAmount: 0,
    totalSeats: seatIds.length,
  };

  seatIds.forEach((seatId) => {
    const category = getSeatCategory(seatId);
    if (breakdown[category]) {
      breakdown[category].count += 1;
      breakdown[category].total += breakdown[category].rate;
      breakdown[category].seats.push(seatId);
      breakdown.totalAmount += breakdown[category].rate;
    }
  });

  return breakdown;
}
