import React, { createContext, useState, useMemo, useCallback } from 'react';

// Step 1: Create the Context
const BookingContext = createContext();

// Movie data lives here so context can filter it
const MOVIES = [
  { id: 1, title: 'Leo',    genre: 'Action',  year: 2023 },
  { id: 2, title: 'GOAT',   genre: 'Action',  year: 2024 },
  { id: 3, title: 'Dragon', genre: 'Fantasy', year: 2025 },
  { id: 4, title: 'Coolie', genre: 'Action',  year: 2025 },
];

const TICKET_PRICE = 150;

// Step 2: Create the Provider component
function BookingProvider({ children }) {
  // All booking state is managed here
  const [searchQuery, setSearchQuery]           = useState('');
  const [selectedMovie, setSelectedMovie]       = useState(null);
  const [selectedTheatre, setSelectedTheatre]   = useState(null);
  const [selectedTime, setSelectedTime]         = useState(null);
  const [selectedSeats, setSelectedSeats]       = useState([]);
  const [bookingConfirmed, setBookingConfirmed] = useState(false);

  // useMemo - filter movies by search query
  const filteredMovies = useMemo(() => {
    if (searchQuery === '') return MOVIES;
    return MOVIES.filter((movie) =>
      movie.title.toLowerCase().includes(searchQuery.toLowerCase())
    );
  }, [searchQuery]);

  // useMemo - calculate total amount
  const totalAmount = useMemo(() => {
    return selectedSeats.length * TICKET_PRICE;
  }, [selectedSeats]);

  // useCallback - select a movie (resets lower selections)
  const handleSelectMovie = useCallback((movie) => {
    setSelectedMovie(movie);
    setSelectedTheatre(null);
    setSelectedTime(null);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  // useCallback - select a theatre
  const handleSelectTheatre = useCallback((theatre) => {
    setSelectedTheatre(theatre);
    setSelectedTime(null);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  // useCallback - select a show time
  const handleSelectTime = useCallback((timing) => {
    setSelectedTime(timing);
    setSelectedSeats([]);
    setBookingConfirmed(false);
  }, []);

  // useCallback - toggle seat (select or remove)
  const handleToggleSeat = useCallback((seat) => {
    setSelectedSeats((prevSeats) => {
      if (prevSeats.includes(seat)) {
        return prevSeats.filter((s) => s !== seat);
      } else {
        return [...prevSeats, seat];
      }
    });
    setBookingConfirmed(false);
  }, []);

  // useCallback - update search query
  const handleSearchChange = useCallback((value) => {
    setSearchQuery(value);
  }, []);

  // Confirm booking
  const handleConfirmBooking = useCallback(() => {
    setBookingConfirmed(true);
  }, []);

  // Step 3: All values and functions shared via context
  const contextValue = {
    searchQuery,
    selectedMovie,
    selectedTheatre,
    selectedTime,
    selectedSeats,
    bookingConfirmed,
    filteredMovies,
    totalAmount,
    handleSearchChange,
    handleSelectMovie,
    handleSelectTheatre,
    handleSelectTime,
    handleToggleSeat,
    handleConfirmBooking,
  };

  return (
    <BookingContext.Provider value={contextValue}>
      {children}
    </BookingContext.Provider>
  );
}

export { BookingContext, BookingProvider };

FILE 2: src/hooks/useBooking.js

import { useContext } from 'react';
import { BookingContext } from '../context/BookingContext';

// Custom Hook - useBooking
// Any component can call this hook to get booking data and functions
function useBooking() {
  const context = useContext(BookingContext);

  if (!context) {
    throw new Error('useBooking must be used within a BookingProvider');
  }

  return context;
}

export default useBooking;

import React, { useRef } from 'react';

// Context
import { BookingProvider } from './context/BookingContext';

// Custom Hook
import useBooking from './hooks/useBooking';

// Components
import Header from './components/Header';
import Home from './components/Home';
import SearchBar from './components/SearchBar';
import MovieList from './components/MovieList';
import TheatreList from './components/TheatreList';
import ShowTiming from './components/ShowTiming';
import SeatSelection from './components/SeatSelection';
import BookingSummary from './components/BookingSummary';
import Footer from './components/Footer';

import './App.css';

// Inner component - can use useBooking hook because it is inside BookingProvider
function AppContent() {
  const searchInputRef = useRef(null);
  const { searchQuery, handleSearchChange } = useBooking();

  return (
    <div className="app-container">
      <Header />

      <main className="main-content">
        <Home />

        <SearchBar
          searchQuery={searchQuery}
          onSearchChange={handleSearchChange}
          searchInputRef={searchInputRef}
        />

        <MovieList />
        <TheatreList />
        <ShowTiming />
        <SeatSelection />
        <BookingSummary />
      </main>

      <Footer />
    </div>
  );
}

// Main App - wraps everything with BookingProvider
function App() {
  return (
    <BookingProvider>
      <AppContent />
    </BookingProvider>
  );
}

export default App;

import React from 'react';
import useBooking from '../hooks/useBooking';

function MovieList() {
  const { filteredMovies, selectedMovie, handleSelectMovie } = useBooking();

  return (
    <div className="section-card">
      <h3 className="section-title">🎬 Select a Movie</h3>

      {filteredMovies.length === 0 && (
        <p style={{ color: '#888', fontStyle: 'italic' }}>No movies found matching your search.</p>
      )}

      <div className="movies-grid">
        {filteredMovies.map((movie) => {
          const isSelected = selectedMovie?.id === movie.id;
          return (
            <div
              key={movie.id}
              onClick={() => handleSelectMovie(movie)}
              className={`movie-card ${isSelected ? 'selected' : ''}`}
            >
              <div className="movie-title">{movie.title}</div>
              <div className="movie-info">{movie.genre} • {movie.year}</div>
              {isSelected && <span className="movie-badge">✓ Selected</span>}
            </div>
          );
        })}
      </div>
    </div>
  );
}

export default MovieList;

import React from 'react';
import useBooking from '../hooks/useBooking';

function BookingSummary() {
  const {
    selectedMovie,
    selectedTheatre,
    selectedTime,
    selectedSeats,
    totalAmount,
    bookingConfirmed,
    handleConfirmBooking,
  } = useBooking();

  if (!selectedMovie) return null;

  const isReady = selectedMovie && selectedTheatre && selectedTime && selectedSeats.length > 0;

  return (
    <div className="section-card">
      <h3 className="section-title">📋 Booking Summary</h3>

      <table className="summary-table">
        <tbody>
          <tr>
            <td className="label">Movie</td>
            <td><strong>{selectedMovie?.title || '—'}</strong></td>
          </tr>
          <tr>
            <td className="label">Theatre</td>
            <td>{selectedTheatre?.name ? `${selectedTheatre.name} (${selectedTheatre.location})` : '—'}</td>
          </tr>
          <tr>
            <td className="label">Show Time</td>
            <td>{selectedTime?.time || '—'}</td>
          </tr>
          <tr>
            <td className="label">Selected Seats</td>
            <td>
              {selectedSeats.length > 0 ? (
                <span style={{ color: '#e50914', fontWeight: 'bold' }}>
                  {selectedSeats.join(', ')}
                </span>
              ) : '—'}
            </td>
          </tr>
          <tr>
            <td className="label">Total Tickets</td>
            <td>{selectedSeats.length}</td>
          </tr>
          <tr>
            <td className="total-label">Total Amount</td>
            <td className="total-value">Rs. {totalAmount}</td>
          </tr>
        </tbody>
      </table>

      <button
        onClick={handleConfirmBooking}
        disabled={!isReady}
        className="confirm-btn"
      >
        Confirm Booking
      </button>

      {!isReady && (
        <p className="confirm-hint">
          * Please select movie, theatre, show time, and at least one seat to confirm.
        </p>
      )}

      {bookingConfirmed && (
        <div className="confirmation-card">
          <h4>🎉 Booking Confirmed Successfully!</h4>
          <p>
            <strong>Movie:</strong> {selectedMovie.title} | <strong>Theatre:</strong> {selectedTheatre?.name}
          </p>
          <p>
            <strong>Time:</strong> {selectedTime?.time} | <strong>Seats:</strong> {selectedSeats.join(', ')}
          </p>
          <p>
            <strong>Total Paid:</strong> Rs. {totalAmount}
          </p>
          <p style={{ marginTop: '8px', color: '#15803d', fontWeight: 'bold' }}>
            Enjoy the show! 🍿
          </p>
        </div>
      )}
    </div>
  );
}

export default BookingSummary;
