```javascript
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';

/**
 * Mock REST API Server Plugin for Unit 4 Experiment 3
 * Handles real HTTP GET, POST, PUT, DELETE requests on '/api/bookings'
 */
function mockApiPlugin() {
  let inMemoryBookings = [];

  const parseBody = (req) => {
    return new Promise((resolve) => {
      let body = '';
      req.on('data', (chunk) => {
        body += chunk;
      });
      req.on('end', () => {
        try {
          resolve(body ? JSON.parse(body) : {});
        } catch {
          resolve({});
        }
      });
    });
  };

  return {
    name: 'mock-booking-api',
    configureServer(server) {
      server.middlewares.use(async (req, res, next) => {
        if (!req.url.startsWith('/api/bookings')) {
          return next();
        }

        const urlParts = req.url.split('?')[0].split('/');
        const bookingId = urlParts.length > 3 ? decodeURIComponent(urlParts[3]) : null;

        res.setHeader('Content-Type', 'application/json');

        // 1. GET /api/bookings (or /api/bookings/:id)
        if (req.method === 'GET') {
          if (bookingId) {
            const found = inMemoryBookings.find((b) => b.bookingId === bookingId || b.reference === bookingId);
            if (!found) {
              res.statusCode = 404;
              return res.end(JSON.stringify({ error: `Booking ${bookingId} not found` }));
            }
            res.statusCode = 200;
            return res.end(JSON.stringify(found));
          }
          res.statusCode = 200;
          return res.end(JSON.stringify(inMemoryBookings));
        }

        // 2. POST /api/bookings
        if (req.method === 'POST') {
          const body = await parseBody(req);
          const newBooking = {
            ...body,
            status: 'CONFIRMED',
            createdAt: new Date().toISOString(),
          };
          inMemoryBookings.unshift(newBooking);
          res.statusCode = 201;
          return res.end(JSON.stringify(newBooking));
        }

        // 3. PUT /api/bookings/:id
        if (req.method === 'PUT') {
          const body = await parseBody(req);
          const idx = inMemoryBookings.findIndex((b) => b.bookingId === bookingId || b.reference === bookingId);
          if (idx === -1) {
            inMemoryBookings.unshift(body);
            res.statusCode = 200;
            return res.end(JSON.stringify(body));
          }
          inMemoryBookings[idx] = {
            ...inMemoryBookings[idx],
            ...body,
            status: 'MODIFIED',
            updatedAt: new Date().toISOString(),
          };
          res.statusCode = 200;
          return res.end(JSON.stringify(inMemoryBookings[idx]));
        }

        // 4. DELETE /api/bookings/:id
        if (req.method === 'DELETE') {
          const idx = inMemoryBookings.findIndex((b) => b.bookingId === bookingId || b.reference === bookingId);
          if (idx !== -1) {
            inMemoryBookings[idx].status = 'CANCELLED';
            inMemoryBookings[idx].cancelledAt = new Date().toISOString();
          }
          res.statusCode = 200;
          return res.end(JSON.stringify({ success: true, bookingId, message: 'Booking cancelled' }));
        }

        next();
      });
    },
  };
}

export default defineConfig({
```

```javascript
/**
 * bookingApiService.js
 * Unit 4 - Experiment 3: Movie Booking REST API Service (CRUD Operations)
 *
 * Implements standard RESTful API calls using Fetch:
 *  - GET    /api/bookings       -> Retrieve booking history
 *  - GET    /api/bookings/:id   -> Retrieve a single booking by ID
 *  - POST   /api/bookings       -> Create a new booking
 *  - PUT    /api/bookings/:id   -> Modify an existing booking (showtime, date, seats)
 *  - DELETE /api/bookings/:id   -> Cancel an existing booking
 *  - GET    /api/shows          -> Retrieve cinema show information
 *
 * Includes network error handling, status code validation, loading states,
 * and reliable persistent storage synchronization.
 */

import { calculateSeatsPriceBreakdown } from '../data/seatLayout.js';

const API_BASE = '/api/bookings';
const STORAGE_KEY_BOOKINGS = 'bmm_my_bookings_list';
const STORAGE_KEY_CONFIRMED = 'bmm_confirmed_show_seats';

// Helper: Read local persistent bookings
function getLocalBookings() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY_BOOKINGS);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

// Helper: Save local persistent bookings
function saveLocalBookings(list) {
  try {
    localStorage.setItem(STORAGE_KEY_BOOKINGS, JSON.stringify(list));
  } catch (err) {
    console.warn('Storage write failed in bookingApiService:', err);
  }
}

// Helper: Generate unique reference ID
function generateBookingId() {
  const chars = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ';
  let ref = 'BMM-CBE-';
  for (let i = 0; i < 6; i++) {
    ref += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return ref;
}

/**
 * a) & e) GET: Fetch Booking History
 * Returns array of bookings with status, movie, theatre, showtime, and seats.
 */
export async function getBookingHistory() {
  try {
    const response = await fetch(API_BASE, {
      method: 'GET',
      headers: { 'Accept': 'application/json' },
    });

    if (response.ok) {
      const data = await response.json();
      return Array.isArray(data) ? data : getLocalBookings();
    }
  } catch (netErr) {
    // If backend endpoint is offline, serve from local storage cache
    console.info('[bookingApiService] Serving GET bookings from persistent cache:', netErr.message);
  }

  // Fallback to local storage
  return getLocalBookings();
}

/**
 * GET: Fetch a single booking by reference ID
 */
export async function getBookingById(bookingId) {
  if (!bookingId) {
    throw new Error('Booking ID is required.');
  }

  try {
    const response = await fetch(`${API_BASE}/${encodeURIComponent(bookingId)}`, {
      method: 'GET',
      headers: { 'Accept': 'application/json' },
    });

    if (response.ok) {
      return await response.json();
    }
  } catch (err) {
    console.info('[bookingApiService] GET single booking fallback:', err.message);
  }

  const list = getLocalBookings();
  const found = list.find((b) => b.bookingId === bookingId || b.reference === bookingId);
  if (!found) {
    throw new Error(`Booking with ID "${bookingId}" not found.`);
  }
  return found;
}

/**
 * b) POST: Create a new movie ticket booking
 */
export async function createBooking(bookingData) {
  if (!bookingData.movie || !bookingData.theatre || !bookingData.date || !bookingData.showtime || !bookingData.seats?.length) {
    throw new Error('Incomplete booking payload. Movie, theatre, date, showtime, and seats are required.');
  }

  const newId = generateBookingId();
  const pricing = calculateSeatsPriceBreakdown(bookingData.seats);

  const payload = {
    bookingId: newId,
    reference: newId,
    movie: {
      id: bookingData.movie.id,
      title: bookingData.movie.title,
      poster: bookingData.movie.poster,
      category: bookingData.movie.category,
      language: bookingData.movie.language,
      year: bookingData.movie.year,
    },
    theatre: {
      id: bookingData.theatre.id,
      name: bookingData.theatre.name,
      area: bookingData.theatre.area,
      address: bookingData.theatre.address,
      screens: bookingData.theatre.screens,
    },
    date: {
      iso: bookingData.date.iso || bookingData.date,
      displayLabel: bookingData.date.displayLabel || bookingData.date,
      full: bookingData.date.full || bookingData.date,
    },
    showtime: {
      id: bookingData.showtime.id || 'st',
      time: bookingData.showtime.time || bookingData.showtime,
      label: bookingData.showtime.label || 'Show',
    },
    seats: [...bookingData.seats].sort(),
    totalSeats: bookingData.seats.length,
    pricing,
    amount: pricing.totalAmount || bookingData.amount || 0,
    status: 'CONFIRMED',
    createdAt: new Date().toISOString(),
    confirmedAt: new Date().toLocaleTimeString('en-IN', {
      hour: '2-digit',
      minute: '2-digit',
      hour12: true,
    }),
  };

  try {
    const response = await fetch(API_BASE, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json',
      },
      body: JSON.stringify(payload),
    });

    if (response.ok) {
      const created = await response.json();
      // Keep local persistent copy in sync
      const list = getLocalBookings();
      saveLocalBookings([created, ...list]);
      return created;
    }
  } catch (err) {
    console.info('[bookingApiService] POST fetch fallback to local store:', err.message);
  }

  // Fallback: save to local list
  const list = getLocalBookings();
  const updatedList = [payload, ...list];
  saveLocalBookings(updatedList);
  return payload;
}

/**
 * c) PUT/PATCH: Modify booking details (e.g., change showtime, date, or seats)
 */
export async function updateBooking(bookingId, updates) {
  if (!bookingId) {
    throw new Error('Booking ID is required to update a booking.');
  }

  let existing = null;
  const list = getLocalBookings();
  const index = list.findIndex((b) => b.bookingId === bookingId || b.reference === bookingId);
  if (index !== -1) {
    existing = list[index];
  }

  if (!existing) {
    throw new Error(`Booking ${bookingId} not found.`);
  }

  if (existing.status === 'CANCELLED') {
    throw new Error('Cannot modify a cancelled booking.');
  }

  // Recalculate price if seats were modified
  const newSeats = updates.seats ? [...updates.seats].sort() : existing.seats;
  const pricing = calculateSeatsPriceBreakdown(newSeats);

  const updatedRecord = {
    ...existing,
    ...updates,
    seats: newSeats,
    totalSeats: newSeats.length,
    pricing,
    amount: pricing.totalAmount,
    status: 'MODIFIED',
    updatedAt: new Date().toISOString(),
    modifiedAt: new Date().toLocaleTimeString('en-IN', {
      hour: '2-digit',
      minute: '2-digit',
      hour12: true,
    }),
  };

  try {
    const response = await fetch(`${API_BASE}/${encodeURIComponent(bookingId)}`, {
      method: 'PUT',
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json',
      },
      body: JSON.stringify(updatedRecord),
    });

    if (response.ok) {
      const saved = await response.json();
      list[index] = saved;
      saveLocalBookings(list);
      return saved;
    }
  } catch (err) {
    console.info('[bookingApiService] PUT fallback to local store:', err.message);
  }

  list[index] = updatedRecord;
  saveLocalBookings(list);
  return updatedRecord;
}

/**
 * d) DELETE: Cancel an existing booking
 */
export async function cancelBooking(bookingId) {
  if (!bookingId) {
    throw new Error('Booking ID is required to cancel a booking.');
  }

  const list = getLocalBookings();
  const index = list.findIndex((b) => b.bookingId === bookingId || b.reference === bookingId);

  if (index === -1) {
    throw new Error(`Booking ${bookingId} not found.`);
  }

  const existing = list[index];

  // Try server DELETE request
  try {
    const response = await fetch(`${API_BASE}/${encodeURIComponent(bookingId)}`, {
      method: 'DELETE',
      headers: { 'Accept': 'application/json' },
    });

    if (response.ok) {
      // Mark as cancelled
      existing.status = 'CANCELLED';
      existing.cancelledAt = new Date().toISOString();
      list[index] = existing;
      saveLocalBookings(list);
      return { success: true, bookingId, message: 'Booking cancelled successfully.' };
    }
  } catch (err) {
    console.info('[bookingApiService] DELETE fallback to local store:', err.message);
  }

  // Mark as cancelled in local storage
  existing.status = 'CANCELLED';
  existing.cancelledAt = new Date().toISOString();
  list[index] = existing;
  saveLocalBookings(list);

  return { success: true, bookingId, message: 'Booking cancelled successfully.' };
}
```

```jsx
import React, { useState, useEffect, useCallback } from 'react';
import {
  getBookingHistory,
  updateBooking,

const SHOWTIMED_OPTIONS = [
  { id: 'st1000', time: '10:00 AM', label: 'Morning Show' },
  { id: 'st1330', time: '1:30 PM',  label: 'Matinee Show' },
  { id: 'st1630', time: '4:30 PM',  label: 'Evening Show' },
  { id: 'st1930', time: '7:30 PM',  label: 'Night Show' },
  { id: 'st2215', time: '10:15 PM', label: 'Late Night' },
];

function BookingHistory() {
  const [bookings, setBookings] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  // Edit Modal State (PUT)
  const [editingBooking, setEditingBooking] = useState(null);
  const [editShowtime, setEditShowtime] = useState('');
  const [editSeatsInput, setEditSeatsInput] = useState('');
  const [isUpdating, setIsUpdating] = useState(false);
  const [editError, setEditError] = useState(null);

  // Cancellation State (DELETE)
  const [cancellingId, setCancellingId] = useState(null);
  const [actionSuccessMsg, setActionSuccessMsg] = useState(null);

  // 1. GET: Fetch Booking History
  const fetchBookings = useCallback(async () => {
    setLoading(true);
    setError(null);
    try {
      const data = await getBookingHistory();
      setBookings(data);
    } catch (err) {
      setError(err.message || 'Failed to load booking history.');
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    fetchBookings();
  }, [fetchBookings]);

  // Open Edit Modal (PUT)
  const handleOpenEdit = (booking) => {
    setEditingBooking(booking);
    setEditShowtime(booking.showtime?.time || '');
    setEditSeatsInput((booking.seats || []).join(', '));
    setEditError(null);
  };

  const handleCloseEdit = () => {
    setEditingBooking(null);
    setEditError(null);
  };

  // 2. PUT: Submit Booking Modifications
  const handleSaveEdit = async (e) => {
    e.preventDefault();
    if (!editingBooking) return;

    const parsedSeats = editSeatsInput
      .split(',')
      .map((s) => s.trim().toUpperCase())
      .filter(Boolean);

    if (parsedSeats.length === 0) {
      setEditError('Booking must have at least 1 seat.');
      return;
    }

    setIsUpdating(true);
    setEditError(null);

    try {
      const selectedTimeObj = SHOWTIMED_OPTIONS.find((t) => t.time === editShowtime) || {
        id: 'mod',
        time: editShowtime,
        label: 'Selected Show',
      };

      const updated = await updateBooking(editingBooking.bookingId, {
        showtime: selectedTimeObj,
        seats: parsedSeats,
      });

      // Update local state with modified record
      setBookings((prev) =>
        prev.map((b) => (b.bookingId === updated.bookingId ? updated : b))
      );

      setActionSuccessMsg(`Booking ${updated.bookingId} modified successfully!`);
      setTimeout(() => setActionSuccessMsg(null), 4000);
      handleCloseEdit();
    } catch (err) {
      setEditError(err.message || 'Unable to update booking.');
    } finally {
      setIsUpdating(false);
    }
  };

  // 3. DELETE: Cancel Booking
  const handleCancelBooking = async (bookingId) => {
    const confirmCancel = window.confirm(
      `Are you sure you want to cancel booking ${bookingId}? This action cannot be undone.`
    );
    if (!confirmCancel) return;

    setCancellingId(bookingId);
    try {
      await cancelBooking(bookingId);

      // Update local list to show CANCELLED status
      setBookings((prev) =>
        prev.map((b) =>
          b.bookingId === bookingId
            ? { ...b, status: 'CANCELLED', cancelledAt: new Date().toISOString() }
            : b
        )
      );

      setActionSuccessMsg(`Booking ${bookingId} has been cancelled.`);
      setTimeout(() => setActionSuccessMsg(null), 4000);
    } catch (err) {
      alert(`Failed to cancel booking: ${err.message}`);
    } finally {
      setCancellingId(null);
    }
  };

  return (
    <div className="booking-history-container">
      <div className="history-header-toolbar">
        <div>
          <h2 className="history-title">My Booking History (REST API CRUD)</h2>
          <p className="history-subtitle">
            View, modify, or cancel movie ticket reservations using GET, PUT, and DELETE APIs
          </p>
        </div>
        <button
          type="button"
          onClick={fetchBookings}
          className="btn-secondary refresh-btn"
          disabled={loading}
          title="Refresh booking list (GET)"
        >
          {loading ? 'Fetching...' : '🔄 Refresh List (GET)'}
        </button>
      </div>

      {/* Success Notification Banner */}
      {actionSuccessMsg && (
        <div className="api-toast-success" role="alert">
          ✓ {actionSuccessMsg}
        </div>
      )}

      {/* 1. Loading State */}
      {loading && (
        <div className="api-loading-panel">
          <div className="spinner"></div>
          <p>Retrieving booking records from API...</p>
        </div>

      {/* 2. Error State */}
      {error && !loading && (

          <div className="error-icon">⚠️</div>
          <div className="error-content">
```

**useBooking.js**
```javascript
import { useState, useCallback, useMemo } from 'react';
import { useSelector, useDispatch } from 'react-redux';
import {
  setMovie,
  setTheatre,
  setShowDate,
  setShowTime,
  toggleSeat,
  removeSeat,
  setSearchQuery,
  setTheatreAreaFilter as setReduxTheatreAreaFilter,
  setTheatreSearchQuery as setReduxTheatreSearchQuery,
  confirmBookingSuccess,
  setBookingError,
  clearBooking,
} from '../redux/bookingSlice';
import { COIMBATORE_THEATRES } from '../data/theatres.js';
import { getAvailableBookingDates } from '../data/showtimes.js';
import { ROW_CONFIG, SEAT_PRICES, TOTAL_SEATS_PER_SCREEN } from '../data/seatLayout.js';
import {
  buildShowKey,
  getBookedSeatsForShow,
  confirmNewBooking,
  getAllSavedBookings,
} from '../services/bookingService.js';
import { createBooking } from '../services/bookingApiService.js';
import { FALLBACK_MOVIES_2026 } from '../data/fallbackMovies.js';

export function useBooking() {
  const dispatch = useDispatch();
  const bookingState = useSelector((state) => state.booking);

  // Available 8 days (Today + 7 days)
  const availableDates = useMemo(() => getAvailableBookingDates(7), []);
  const selectedDate = bookingState.selectedDate || availableDates[0];

  // Saved bookings from localStorage (reactive state)
  const [allBookings, setAllBookings] = useState(() => getAllSavedBookings());

  const lastBooking = useMemo(() => {
    return allBookings && allBookings.length > 0 ? allBookings[0] : null;
  }, [allBookings]);

  // Current screening key based on Redux selection
  const currentShowKey = useMemo(() => {
    if (
      !bookingState.selectedMovie ||
      !bookingState.selectedTheatre ||
      !selectedDate ||
      !bookingState.selectedTime
    ) {
      return null;
    }
    return buildShowKey(
      bookingState.selectedMovie.id,
      bookingState.selectedTheatre.id,
      selectedDate.iso || selectedDate,
      bookingState.selectedTime.id || bookingState.selectedTime.time
    );
  }, [
    bookingState.selectedMovie,
    bookingState.selectedTheatre,
    selectedDate,
    bookingState.selectedTime,
  ]);

  // Booked seats for current show
  const bookedSeatsForCurrentShow = useMemo(() => {
    if (!currentShowKey) return [];
    return getBookedSeatsForShow(currentShowKey, bookingState.bookingConfirmed);
  }, [currentShowKey, bookingState.bookingConfirmed]);

  // Filter movies by search query
  const filteredMovies = useMemo(() => {
    const q = (bookingState.searchQuery || '').trim().toLowerCase();
    if (!q) return FALLBACK_MOVIES_2026;
    return FALLBACK_MOVIES_2026.filter((movie) =>
      movie.title.toLowerCase().includes(q)
    );
  }, [bookingState.searchQuery]);

  // Redux Action Dispatchers
  const handleSelectMovie = useCallback(
    (movie) => {
      dispatch(setMovie(movie));
    },
    [dispatch]
  );

  const handleSelectTheatre = useCallback(
    (theatre) => {
      dispatch(setTheatre(theatre));
    },
    [dispatch]
  );

  const handleSelectDate = useCallback(
    (date) => {
      dispatch(setShowDate(date));
    },
    [dispatch]
  );

  const handleSelectTime = useCallback(
    (timing) => {
      dispatch(setShowTime(timing));
    },
    [dispatch]
  );

  const handleToggleSeat = useCallback(
    (seatId) => {
      if (bookedSeatsForCurrentShow.includes(seatId)) {
        return false;
      }
      dispatch(toggleSeat(seatId));
      return true;
    },
    [dispatch, bookedSeatsForCurrentShow]
  );

  const handleRemoveSeat = useCallback(
    (seatId) => {
      dispatch(removeSeat(seatId));
    },
    [dispatch]
  );

  const handleSearchChange = useCallback(
    (value) => {
      dispatch(setSearchQuery(value));
    },
    [dispatch]
  );

  const setTheatreAreaFilter = useCallback(
    (area) => {
      dispatch(setReduxTheatreAreaFilter(area));
    },
    [dispatch]
  );

  const setTheatreSearchQuery = useCallback(
    (query) => {
      dispatch(setReduxTheatreSearchQuery(query));
    },
    [dispatch]
  );

  const handleConfirmBooking = useCallback(async () => {
    if (
      !bookingState.selectedMovie ||
      !bookingState.selectedTheatre ||
      !selectedDate ||
      !bookingState.selectedTime ||
      bookingState.selectedSeats.length === 0
    ) {
      dispatch(
        setBookingError('Please complete all booking steps (Theatre, Date, Showtime, and at least 1 Seat).')
      );
      return false;
    }

    try {
      const record = confirmNewBooking({
        movie: bookingState.selectedMovie,
        theatre: bookingState.selectedTheatre,
        date: selectedDate,
        showtime: bookingState.selectedTime,
        seats: bookingState.selectedSeats,
      });

      // POST: Send new booking to REST API
      try {
        await createBooking(record);
      } catch (apiErr) {
        console.info('POST /api/bookings fallback synced:', apiErr.message);
      }

      dispatch(confirmBookingSuccess(record.bookingId));
      setAllBookings(getAllSavedBookings());
      return true;
    } catch (err) {
      dispatch(
        setBookingError(err.message || 'Unable to confirm booking. Please try again.')
      );
      return false;
    }
  }, [
    dispatch,
    bookingState.selectedMovie,
    bookingState.selectedTheatre,
    selectedDate,
    bookingState.selectedTime,
    bookingState.selectedSeats,
  ]);

  const handleResetBooking = useCallback(() => {
    dispatch(clearBooking());
  }, [dispatch]);

  return {
    selectedMovie: bookingState.selectedMovie,
    selectedTheatre: bookingState.selectedTheatre,
    selectedDate,
    selectedTime: bookingState.selectedTime,
    selectedSeats: bookingState.selectedSeats,
    totalAmount: bookingState.totalAmount,
    totalSeats: bookingState.totalSeats,
    priceBreakdown: bookingState.priceBreakdown,
    bookingConfirmed: bookingState.bookingConfirmed,
    activeBookingReference: bookingState.bookingReference,
    bookingError: bookingState.bookingError,
    searchQuery: bookingState.searchQuery,
    theatreAreaFilter: bookingState.theatreAreaFilter || 'All Areas',
    theatreSearchQuery: bookingState.theatreSearchQuery || '',

    theatres: COIMBATORE_THEATRES,
    availableDates,
    currentShowKey,
    bookedSeatsForCurrentShow,
    filteredMovies,
    seatPrices: SEAT_PRICES,
    ticketPrice: SEAT_PRICES.Regular,
    rowConfig: ROW_CONFIG,
    totalSeatsPerScreen: TOTAL_SEATS_PER_SCREEN,
    allBookings,
    lastBooking,

    handleSelectMovie,
    handleSelectTheatre,
    handleSelectDate,
    handleSelectTime,
    handleToggleSeat,
    handleRemoveSeat,
    handleSearchChange,
    handleConfirmBooking,
    handleResetBooking,
    setTheatreAreaFilter,
    setTheatreSearchQuery,
  };
}

export default useBooking;
```
