```javascript
              ...body,
              status: 'CONFIRMED',
```

```javascript
import axios from 'axios';

export const TOKEN_KEY = 'bmm_auth_token';
export const USER_KEY = 'bmm_auth_user';

export const getToken = () => {
  if (typeof window === 'undefined') return null;
  return sessionStorage.getItem(TOKEN_KEY) || localStorage.getItem(TOKEN_KEY);
};

export const setToken = (token, persist = false) => {
  if (typeof window === 'undefined') return;
  sessionStorage.setItem(TOKEN_KEY, token);
  if (persist) {
    localStorage.setItem(TOKEN_KEY, token);
  }
};

export const removeToken = () => {
  if (typeof window === 'undefined') return;
  sessionStorage.removeItem(TOKEN_KEY);
  sessionStorage.removeItem(USER_KEY);
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(USER_KEY);
};

export const getUser = () => {
  if (typeof window === 'undefined') return null;
  try {
    const raw = sessionStorage.getItem(USER_KEY) || localStorage.getItem(USER_KEY);
    return raw ? JSON.parse(raw) : null;
  } catch {
    return null;
  }
};

export const setUser = (user, persist = false) => {
  if (typeof window === 'undefined') return;
  sessionStorage.setItem(USER_KEY, JSON.stringify(user));
  if (persist) {
    localStorage.setItem(USER_KEY, JSON.stringify(user));
  }
};

/**
 * Creates an expired mock JWT token to explicitly demonstrate
 * Experiment 4 requirement (g): Handling expired tokens & 401 errors.
 */
export const injectExpiredTokenForTesting = () => {
  const expiredPayload = {
    username: 'test_user',
    email: 'test@foodorder.com',
    role: 'user',
    exp: Math.floor(Date.now() / 1000) - 1800,
  };
  const expiredToken = 'bmm_jwt_' + btoa(JSON.stringify(expiredPayload));
  setToken(expiredToken);
  return expiredToken;
};

// Create configured Axios instance
const axiosInstance = axios.create({
  baseURL: '',
  headers: {
    'Content-Type': 'application/json',
    Accept: 'application/json',
  },
  timeout: 10000,
});

/**
 * Request Interceptor: Attach authentication token to outgoing requests
 * Requirement (d): Attach the token to API requests using Axios configuration or interceptors.
 */
axiosInstance.interceptors.request.use(
  (config) => {
    const token = getToken();
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

/**
 * Response Interceptor: Intercept 401/403 status responses
 * Requirement (g): Handle authentication errors and expired tokens.
 */
axiosInstance.interceptors.response.use(
  (response) => response,
  (error) => {
    const status = error.response ? error.response.status : null;
    if (status === 401 || status === 403) {
      removeToken();

      if (typeof window !== 'undefined') {
        const errorDetail = error.response?.data || {
          message: 'Session has expired or authentication token is missing. Please sign in again.',
        };

        window.dispatchEvent(
          new CustomEvent('bmm_auth_unauthorized', { detail: errorDetail })
        );

        if (!window.location.pathname.startsWith('/login')) {
          const redirectPath = encodeURIComponent(
            window.location.pathname + window.location.search
          );
          window.location.href = `/login?redirect=${redirectPath}&expired=true`;
        }
      }
    }
    return Promise.reject(error);
  }
);

export default axiosInstance;
```

```javascript
import axiosInstance, {
  getToken,
  setToken,
  getUser,
  setUser,
  removeToken,
} from './axiosInstance';

const AUTH_API = '/api/auth';

/**
 * Authentication Service for Unit 4 - Experiment 4
 * Communicates with the authentication API using Axios and manages session tokens.
 */
export const authService = {
  /**
   * Send login credentials to Authentication API
   * Requirement (b): Send login credentials to an authentication API.
   * Requirement (c): Store the authentication token after successful login.
   */
  async login(credentials) {
    const usernameOrEmail = credentials.username || credentials.email || '';
    const password = credentials.password || '';

    try {
      const response = await axiosInstance.post(`${AUTH_API}/login`, {
        username: usernameOrEmail,
        password,
      });

      const { token, user } = response.data;
      if (token) {
        setToken(token);
      }
      if (user) {
        setUser(user);
      }

      return {
        success: true,
        token,
        user: user || { username: usernameOrEmail },
      };
    } catch (error) {
      if (!error.response && (error.code === 'ERR_NETWORK' || error.message?.includes('Network Error'))) {
        console.warn('[authService] Backend offline - running local mock authentication fallback');
        if (password.length >= 4) {
          const fallbackUser = {
            username: usernameOrEmail.includes('@') ? usernameOrEmail.split('@')[0] : usernameOrEmail,
            email: usernameOrEmail.includes('@') ? usernameOrEmail : `${usernameOrEmail}@bookmymovie.com`,
            role: 'user',
          };
          const mockToken =
            'bmm_jwt_' +
            btoa(
              JSON.stringify({
                username: fallbackUser.username,
                exp: Math.floor(Date.now() / 1000) + 3600,
              })
            );

          setToken(mockToken);
          setUser(fallbackUser);
          return { success: true, token: mockToken, user: fallbackUser };
        }
        return { success: false, message: 'Invalid credentials. Password must be at least 4 characters.' };
      }

      const message =
        error.response?.data?.error ||
        error.response?.data?.message ||
        error.message ||
        'Authentication failed. Please check your credentials.';
      return { success: false, message };
    }
  },

  /**
   * Log out user and clean up session tokens
   * Requirement (e): Implement logout functionality and remove the stored token.
   */
  async logout() {
    try {
      await axiosInstance.post(`${AUTH_API}/logout`);
    } catch {
      // Best-effort server notification
    } finally {
      removeToken();
    }
    return { success: true };
  },

  async getCurrentUser() {
    const token = getToken();
    if (!token) return null;

    try {
      const response = await axiosInstance.get(`${AUTH_API}/me`);
      return response.data?.user || getUser();
    } catch {
      return getUser();
    }
  },

  getToken,
  getUser,
  removeToken,
};

export default authService;
```

```jsx
import React, { createContext, useState, useEffect, useCallback } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import authService from '../services/authService';
import {
  loginStart,
  loginSuccess,
  loginFailure,
  logoutSuccess,
  clearAuthError,
} from '../redux/authSlice';
import { clearBooking } from '../redux/bookingSlice';

const AuthContext = createContext();

export function AuthProvider({ children }) {
  const dispatch = useDispatch();
  const reduxAuth = useSelector((state) => state.auth);

  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  // Synchronize on mount and handle unauthorized 401/403 events from Axios
  useEffect(() => {
    const handleUnauthorized = (e) => {
      const msg = e.detail?.error || e.detail?.message || 'Session expired. Please log in again.';
      dispatch(logoutSuccess());
      dispatch(clearBooking());
      setError(msg);
    };

    window.addEventListener('bmm_auth_unauthorized', handleUnauthorized);
    return () => {
      window.removeEventListener('bmm_auth_unauthorized', handleUnauthorized);
    };
  }, [dispatch]);

  const login = useCallback(
    async (usernameOrEmail, password) => {
      const trimmedId = (usernameOrEmail || '').trim();
      const trimmedPass = (password || '').trim();

      if (!trimmedId || !trimmedPass) {
        const msg = 'Please enter both username/email and password.';
        setError(msg);
        return { success: false, message: msg };
      }

      setLoading(true);
      setError(null);
      dispatch(loginStart());

      try {
        const result = await authService.login({
          username: trimmedId,
          password: trimmedPass,
        });

        if (result.success) {
          dispatch(
            loginSuccess({
              token: result.token,
              user: result.user,
            })
          );
          setLoading(false);
          return { success: true, token: result.token, user: result.user };
        }

        dispatch(loginFailure(result.message));
        setError(result.message);
        setLoading(false);
        return { success: false, message: result.message };
      } catch (err) {
        const msg = err.message || 'Login request failed.';
        dispatch(loginFailure(msg));
        setError(msg);
        setLoading(false);
        return { success: false, message: msg };
      }
    },
    [dispatch]
  );

  const logout = useCallback(async () => {
    setLoading(true);
    await authService.logout();
    dispatch(logoutSuccess());
    dispatch(clearBooking());
    setError(null);
    setLoading(false);
  }, [dispatch]);

  const clearError = useCallback(() => {
    setError(null);
    dispatch(clearAuthError());
  }, [dispatch]);

  const value = {
    token: reduxAuth.token,
    user: reduxAuth.user,
    username: reduxAuth.user?.username || '',
    isAuthenticated: reduxAuth.isAuthenticated,
    loading: loading || reduxAuth.loading,
    error: error || reduxAuth.error,
    login,
    logout,
    clearError,
  };

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export { AuthContext };
export default AuthContext;
```
