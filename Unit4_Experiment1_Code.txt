export function isLiveApiConfigured() {
  return Boolean(
    TMDB_API_KEY &&
    TMDB_API_KEY.trim() !== '' &&
    TMDB_API_KEY !== 'your_tmdb_api_key_here'
  );
}

function normalizeTmdbMovie(movie) {
  if (!movie) return null;

  let genresList = [];
  if (Array.isArray(movie.genres) && movie.genres.length > 0) {
    genresList = movie.genres
      .map((g) => (typeof g === 'object' && g ? g.name : g))
      .filter(Boolean);
  } else if (Array.isArray(movie.genre_ids) && movie.genre_ids.length > 0) {
    genresList = movie.genre_ids.map((id) => TMDB_GENRES[id] || 'General').filter(Boolean);
  }

  if (genresList.length === 0) {
    genresList = ['Cinema'];
  }

  const origLang = movie.original_language || '';
  const isTamil = origLang.toLowerCase() === 'ta';
  const isEnglish = origLang.toLowerCase() === 'en';

  const category = isTamil ? 'Tamil' : isEnglish ? 'Hollywood' : 'International';
  const language = isTamil ? 'Tamil' : isEnglish ? 'English' : origLang.toUpperCase();

  const year = movie.release_date
    ? new Date(movie.release_date).getFullYear()
    : 2026;

  const isUpcoming = movie.release_date
    ? new Date(movie.release_date) > new Date()
    : true;

  const poster = movie.poster_path
    ? `${TMDB_IMAGE_BASE}${movie.poster_path}`
    : '/posters/doomsday.jpg';

  const backdrop = movie.backdrop_path
    ? `${TMDB_BACKDROP_BASE}${movie.backdrop_path}`
    : poster;

  const rating =
    movie.vote_average !== undefined && movie.vote_average !== null && movie.vote_average > 0
      ? Number(movie.vote_average.toFixed(1))
      : 8.5;

  let releaseDateFormatted = movie.release_date || '2026';
  if (movie.release_date) {
    try {
      const d = new Date(movie.release_date);
      if (!isNaN(d)) {
        releaseDateFormatted = d.toLocaleDateString('en-US', {
          month: 'long',
          day: 'numeric',
          year: 'numeric',
        });
      }
    } catch {
      releaseDateFormatted = movie.release_date;
    }
  }

  let cast = 'All-Star Ensemble Cast';
  let director = 'Acclaimed Director';

  if (movie.credits) {
    if (Array.isArray(movie.credits.cast) && movie.credits.cast.length > 0) {
      cast = movie.credits.cast.slice(0, 4).map((c) => c.name).join(', ');
    }
    if (Array.isArray(movie.credits.crew)) {
      const dirObj = movie.credits.crew.find((c) => c.job === 'Director');
      if (dirObj) director = dirObj.name;
    }
  }

  return {
    id: String(movie.id),
    apiId: movie.id,
    source: 'tmdb-api',
    title: movie.title || movie.original_title || 'Untitled',
    category,
    language,
    genre: genresList.join(' / '),
    genres: genresList,
    rating,
    year,
    releaseDate: movie.release_date || '2026',
    releaseDateFormatted,
    status: isUpcoming ? 'Upcoming' : 'Released',
    isUpcoming,
    runtime: movie.runtime ? `${movie.runtime} min` : '150 min',
    description: movie.overview || '2026 cinema presentation. No synopsis provided.',
    poster,
    backdrop,
    cast,
    director,
  };
}

function filterFallbackDataset({ search = '', category = 'All Movies', genre = 'All Genres' }) {
  let list = [...FALLBACK_MOVIES_2026];

  if (category && category !== 'All' && category !== 'All Movies') {
    list = list.filter((m) => m.category.toLowerCase() === category.toLowerCase());
  }

  if (genre && genre !== 'All' && genre !== 'All Genres') {
    const targetGenre = genre.toLowerCase();
    list = list.filter(
      (m) =>
        m.genres.some((g) => g.toLowerCase() === targetGenre) ||
        m.genre.toLowerCase().includes(targetGenre)
    );
  }

  if (search && search.trim() !== '') {
    const q = search.trim().toLowerCase();
    list = list.filter((m) => m.title.toLowerCase().includes(q));
  }

  return list;
}

export async function fetchMovies({
  search = '',
  query = '',
  category = 'All Movies',
  genre = 'All Genres',
  page = 1,
  signal,
} = {}) {
  const searchTerm = search || query || '';
  const trimmedSearch = searchTerm.trim();
  const hasKey = isLiveApiConfigured();

  if (!hasKey) {
    const fallbackResults = filterFallbackDataset({
      search: trimmedSearch,
      category,
      genre,
    });
    return {
      movies: fallbackResults,
      isLiveApi: false,
      total: fallbackResults.length,
      page: 1,
      totalPages: 1,
      source: 'local-fallback-2026',
    };
  }

  try {
    let rawMovies = [];
    let totalPages = 1;

    if (trimmedSearch) {
      const url = `${TMDB_BASE_URL}/search/movie?api_key=${TMDB_API_KEY}&query=${encodeURIComponent(
        trimmedSearch
      )}&primary_release_year=2026&include_adult=false&page=${page}`;
      const response = await fetch(url, { signal });

      if (!response.ok) {
        throw new Error(`TMDB Search error: HTTP ${response.status}`);
      }

      const data = await response.json();
      rawMovies = Array.isArray(data.results) ? data.results : [];
      totalPages = data.total_pages || 1;
    } else {
      let endpoint = `${TMDB_BASE_URL}/discover/movie?api_key=${TMDB_API_KEY}&primary_release_year=2026&sort_by=popularity.desc&page=${page}`;

      if (category === 'Tamil') {
        endpoint = `${TMDB_BASE_URL}/discover/movie?api_key=${TMDB_API_KEY}&primary_release_year=2026&with_original_language=ta&sort_by=popularity.desc&page=${page}`;
      } else if (category === 'Hollywood') {
        endpoint = `${TMDB_BASE_URL}/discover/movie?api_key=${TMDB_API_KEY}&primary_release_year=2026&with_original_language=en&sort_by=popularity.desc&page=${page}`;
      }

      const response = await fetch(endpoint, { signal });

      if (!response.ok) {
        throw new Error(`TMDB Discover error: HTTP ${response.status}`);
      }

      const data = await response.json();
      rawMovies = Array.isArray(data.results) ? data.results : [];
      totalPages = data.total_pages || 1;
    }

    let movies = rawMovies.map(normalizeTmdbMovie).filter(Boolean);

    if (category && category !== 'All' && category !== 'All Movies') {
      movies = movies.filter((m) => m.category.toLowerCase() === category.toLowerCase());
    }

    if (genre && genre !== 'All' && genre !== 'All Genres') {
      const targetGenre = genre.toLowerCase();
      movies = movies.filter(
        (m) =>
          m.genres.some((g) => g.toLowerCase() === targetGenre) ||
          m.genre.toLowerCase().includes(targetGenre)
      );
    }

    return {
      movies,
      isLiveApi: true,
      total: movies.length,
      page,
      totalPages,
      source: 'tmdb-api',
    };
  } catch (error) {
    if (error.name === 'AbortError') throw error;

    console.warn('TMDB API fetch failed. Switching to local 2026 fallback dataset:', error.message);

    const fallbackResults = filterFallbackDataset({
      search: trimmedSearch,
      category,
      genre,
    });

    return {
      movies: fallbackResults,
      isLiveApi: false,
      total: fallbackResults.length,
      page: 1,
      totalPages: 1,
      source: 'local-fallback-2026',
      fallbackReason: error.message,
    };
  }
}

export async function getMovieById(id, { signal } = {}) {
  if (!id) {
    throw new Error('Movie ID is required');
  }

  const strId = String(id).trim();

  if (strId.startsWith('fb-') || strId.startsWith('bmm-')) {
    const fallbackMatch = getFallbackMovieById(strId);
    if (fallbackMatch) return fallbackMatch;
  }

  const exactLocalMatch = FALLBACK_MOVIES_2026.find((m) => String(m.id) === strId);
  if (exactLocalMatch) return exactLocalMatch;

  const isNumeric = /^\d+$/.test(strId);
  const numVal = parseInt(strId, 10);

  if (isNumeric && numVal >= 1 && numVal <= FALLBACK_MOVIES_2026.length) {
    if (!isLiveApiConfigured() || numVal <= 15) {
      const indexedMovie = FALLBACK_MOVIES_2026[numVal - 1];
      if (indexedMovie) return indexedMovie;
    }
  }

  if (isNumeric && isLiveApiConfigured()) {
    try {
      const url = `${TMDB_BASE_URL}/movie/${strId}?api_key=${TMDB_API_KEY}&append_to_response=credits`;
      const response = await fetch(url, { signal });

      if (response.ok) {
        const movieData = await response.json();
        return normalizeTmdbMovie(movieData);
      }
    } catch (error) {
      if (error.name === 'AbortError') throw error;
    }
  }

  const cleanQuery = strId.toLowerCase().replace(/[^a-z0-9]/g, '');
  if (cleanQuery) {
    const slugMatch = FALLBACK_MOVIES_2026.find((m) => {
      const cleanTitle = m.title.toLowerCase().replace(/[^a-z0-9]/g, '');
      const cleanId = String(m.id).toLowerCase().replace(/[^a-z0-9]/g, '');
      return (
        cleanTitle === cleanQuery ||
        cleanTitle.includes(cleanQuery) ||
        cleanQuery.includes(cleanTitle) ||
        cleanId.includes(cleanQuery)
      );
    });

    if (slugMatch) return slugMatch;
  }

  if (FALLBACK_MOVIES_2026.length > 0) {
    return FALLBACK_MOVIES_2026[0];
  }

  throw new Error(`Movie with identifier "${id}" was not found.`);
}

export function getAvailableBookingDates(daysAhead = 7) {
  const dates = [];
  const now = new Date();
  const dayNamesShort = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  const monthNamesShort = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

  for (let i = 0; i <= daysAhead; i++) {
    const d = new Date(now);
    d.setDate(now.getDate() + i);

    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    const iso = `${year}-${month}-${day}`;

    const dayOfWeek = dayNamesShort[d.getDay()];
    const dateNum = d.getDate();
    const monthStr = monthNamesShort[d.getMonth()];
    const displayLabel = `${dayOfWeek}, ${dateNum} ${monthStr}`;

    dates.push({
      id: `date-${iso}`,
      iso,
      dayNum: dateNum,
      dayName: i === 0 ? 'Today' : i === 1 ? 'Tomorrow' : dayOfWeek,
      dayOfWeek,
      monthStr,
      year,
      displayLabel,
      full: d.toLocaleDateString('en-IN', {
        weekday: 'short',
        day: 'numeric',
        month: 'short',
        year: 'numeric',
      }),
      isToday: i === 0,
    });
  }

  return dates;
}

export function isShowtimePast(dateIso, showtime) {
  if (!dateIso || !showtime) return false;

  const now = new Date();
  const todayIso = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(
    now.getDate()
  ).padStart(2, '0')}`;

  if (dateIso > todayIso) {
    return false;
  }

  if (dateIso < todayIso) {
    return true;
  }

  const currentHour = now.getHours();
  const currentMinute = now.getMinutes();

  if (showtime.hour24 < currentHour) return true;
  if (showtime.hour24 === currentHour && showtime.minute <= currentMinute) return true;

  return false;
}

export function getShowtimesForTheatre(theatreId, dateIso) {
  return STANDARD_SHOWTIMES.map((st) => {
    const isPast = isShowtimePast(dateIso, st);
    return {
      ...st,
      isAvailable: !isPast,
      theatreId: theatreId || 'default',
      dateIso: dateIso || '',
    };
  });
}

export function generateScreenSeats() {
  const allSeats = [];
  ROW_CONFIG.forEach(({ row, category, price, seatsCount }) => {
    for (let num = 1; num <= seatsCount; num++) {
      allSeats.push({
        id: `${row}${num}`,
        row,
        number: num,
        category,
        price,
        isAisleLeft: num === 5,
        isAisleRight: num === 6,
      });
    }
  });
  return allSeats;
}

export function getSeatCategory(seatId) {
  if (!seatId) return 'Regular';
  const row = String(seatId).trim().charAt(0).toUpperCase();
  const found = ROW_CONFIG.find((r) => r.row === row);
  return found ? found.category : 'Regular';
}

export function getSeatPrice(seatId) {
  if (!seatId) return SEAT_PRICES.Regular;
  const category = getSeatCategory(seatId);
  return SEAT_PRICES[category] || SEAT_PRICES.Regular;
}

export function calculateSeatsPriceBreakdown(seatIds = []) {
  const breakdown = {
    Regular:  { count: 0, rate: SEAT_PRICES.Regular,  total: 0, seats: [] },
    Premium:  { count: 0, rate: SEAT_PRICES.Premium,  total: 0, seats: [] },
    Recliner: { count: 0, rate: SEAT_PRICES.Recliner, total: 0, seats: [] },
    totalAmount: 0,
    totalSeats: seatIds.length,
  };

  seatIds.forEach((seatId) => {
    const category = getSeatCategory(seatId);
    if (breakdown[category]) {
      breakdown[category].count += 1;
      breakdown[category].total += breakdown[category].rate;
      breakdown[category].seats.push(seatId);
      breakdown.totalAmount += breakdown[category].rate;
    }
  });

  return breakdown;
}

const STORAGE_KEY_CONFIRMED = 'bmm_confirmed_show_seats';
const STORAGE_KEY_BOOKINGS = 'bmm_my_bookings_list';
const STORAGE_KEY_LEGACY = 'bm_last_booking';

export function buildShowKey(movieId, theatreId, dateIso, showtimeId) {
  const m = movieId ? String(movieId).replace(/[^a-zA-Z0-9]/g, '') : 'anyMovie';
  const t = theatreId ? String(theatreId).replace(/[^a-zA-Z0-9]/g, '') : 'anyTheatre';
  const d = dateIso ? String(dateIso).replace(/[^a-zA-Z0-9]/g, '') : 'today';
  const s = showtimeId ? String(showtimeId).replace(/[^a-zA-Z0-9]/g, '') : 'st1000';
  return `${m}_${t}_${d}_${s}`;
}

function hashString(str) {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = (hash << 5) - hash + str.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash);
}

export function getInitialPreBookedSeats(showKey) {
  const seed = hashString(showKey || 'default_cbe_show');
  const bookedSet = new Set();
  const countToBook = 22 + (seed % 12);

  let currentHash = seed;
  let attempts = 0;

  while (bookedSet.size < countToBook && attempts < 200) {
    attempts++;
    currentHash = (currentHash * 9301 + 49297) % 233280;

    const rowIndex = currentHash % ROW_CONFIG.length;
    const rowObj = ROW_CONFIG[rowIndex];

    currentHash = (currentHash * 9301 + 49297) % 233280;
    const seatNum = 1 + (currentHash % rowObj.seatsCount);

    const seatId = `${rowObj.row}${seatNum}`;
    bookedSet.add(seatId);
  }

  if (bookedSet.size < 20) {
    ['A3', 'A4', 'B2', 'B5', 'C4', 'D1', 'D7', 'E3', 'F2', 'F6', 'G5', 'H1', 'I4', 'J8'].forEach((s) =>
      bookedSet.add(s)
    );
  }

  return Array.from(bookedSet);
}

function getConfirmedBookingsMap() {

export function getBookedSeatsForShow(showKey) {
  const initial = getInitialPreBookedSeats(showKey);
  const confirmedMap = getConfirmedBookingsMap();
  const userBooked = confirmedMap[showKey] || [];

  return Array.from(new Set([...initial, ...userBooked]));
}

export function generateBookingReference() {
  const chars = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ';
  let ref = 'BMM-CBE-';
  for (let i = 0; i < 6; i++) {
    ref += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return ref;
}

export function confirmNewBooking({
  movie,
  theatre,
  date,
  showtime,
  seats = [],
}) {
  if (!movie || !theatre || !date || !showtime || seats.length === 0) {
    throw new Error('Incomplete booking details. Movie, theatre, date, showtime, and seats are required.');
  }

  const showKey = buildShowKey(movie.id, theatre.id, date.iso, showtime.id);
  const currentBooked = getBookedSeatsForShow(showKey);

  const conflicts = seats.filter((s) => currentBooked.includes(s));
  if (conflicts.length > 0) {
    throw new Error(`The following seats are already booked for this show: ${conflicts.join(', ')}`);
  }

  const bookingReference = generateBookingReference();
  const pricing = calculateSeatsPriceBreakdown(seats);

  const bookingRecord = {
    bookingId: bookingReference,
    reference: bookingReference,
    showKey,
    movie: {
      id: movie.id,
      title: movie.title,
      poster: movie.poster,
      category: movie.category,
      language: movie.language,
      year: movie.year,
      rating: movie.rating,
      runtime: movie.runtime,
    },
    theatre: {
      id: theatre.id,
      name: theatre.name,
      area: theatre.area,
      address: theatre.address,
      screens: theatre.screens,
      formats: theatre.formats,
    },
    date: {
      iso: date.iso,
      displayLabel: date.displayLabel,
      full: date.full,
    },
    showtime: {
      id: showtime.id,
      time: showtime.time,
      label: showtime.label,
    },
    seats: [...seats].sort(),
    totalSeats: seats.length,
    pricing,
    amount: pricing.totalAmount,
    createdAt: new Date().toISOString(),
    confirmedAt: new Date().toLocaleTimeString('en-IN', {
      hour: '2-digit',
      minute: '2-digit',
      hour12: true,
    }),
  };

  const confirmedMap = getConfirmedBookingsMap();
  const previousShowSeats = confirmedMap[showKey] || [];
  confirmedMap[showKey] = Array.from(new Set([...previousShowSeats, ...seats]));
  saveConfirmedBookingsMap(confirmedMap);

  try {
    if (typeof localStorage !== 'undefined') {
      const rawList = localStorage.getItem(STORAGE_KEY_BOOKINGS);
      const existingList = rawList ? JSON.parse(rawList) : [];
      const updatedList = [bookingRecord, ...existingList];
      localStorage.setItem(STORAGE_KEY_BOOKINGS, JSON.stringify(updatedList));
      localStorage.setItem(STORAGE_KEY_LEGACY, JSON.stringify(bookingRecord));
    }
  } catch (err) {
    console.warn('Failed saving booking history record:', err);
  }

  return bookingRecord;
}

export function getAllSavedBookings() {
  try {
    if (typeof localStorage === 'undefined') return [];
    const rawList = localStorage.getItem(STORAGE_KEY_BOOKINGS);
    if (rawList) {
      return JSON.parse(rawList);
    }

    const legacy = localStorage.getItem(STORAGE_KEY_LEGACY);
    if (legacy) {
      return [JSON.parse(legacy)];
    }

    return [];
  } catch (err) {
    console.warn('Error reading saved bookings:', err);
    return [];
  }
}

const AuthContext = createContext();

const ADMIN_USER = { username: 'admin', password: '1234' };

function getStoredUsers() {

  }

function saveStoredUsers(users) {
  localStorage.setItem(STORAGE_KEY_USERS, JSON.stringify(users));
}

function obfuscate(str) {
  return btoa(unescape(encodeURIComponent(str)));
}

function deobfuscate(str) {
  try {
    return decodeURIComponent(escape(atob(str)));
  } catch {
    return '';
  }
}

function AuthProvider({ children }) {
  const [username, setUsername] = useState(() =>
    localStorage.getItem(STORAGE_KEY_AUTH_USER) || ''
  );

  const [isAuthenticated, setIsAuthenticated] = useState(() =>
    localStorage.getItem(STORAGE_KEY_AUTH_STATUS) === 'true'
  );

  const register = (newUser, newPass, confirmPass) => {
    const trimmedUser = newUser.trim();
    const trimmedPass = newPass.trim();
    const trimmedConfirm = confirmPass.trim();

    if (!trimmedUser || !trimmedPass || !trimmedConfirm) {
      return { success: false, message: 'All fields are required.' };
    }

    if (trimmedUser.length < 3) {
      return { success: false, message: 'Username must be at least 3 characters.' };
    }

    if (trimmedPass.length < 4) {
      return { success: false, message: 'Password must be at least 4 characters.' };
    }

    if (trimmedPass !== trimmedConfirm) {
      return { success: false, message: 'Passwords do not match.' };
    }

    if (trimmedUser.toLowerCase() === ADMIN_USER.username.toLowerCase()) {
      return { success: false, message: 'That username is already taken. Please choose another.' };
    }

    const users = getStoredUsers();
    const exists = users.some(
      (u) => u.username.toLowerCase() === trimmedUser.toLowerCase()
    );

    if (exists) {
      return { success: false, message: 'That username is already registered. Please log in.' };
    }

    users.push({ username: trimmedUser, password: obfuscate(trimmedPass) });
    saveStoredUsers(users);

    return { success: true };
  };

  const login = (inputUser, inputPass) => {
    const trimmedUser = inputUser.trim();
    const trimmedPass = inputPass.trim();

    if (!trimmedUser || !trimmedPass) {
      return { success: false, message: 'Please enter both username and password.' };
    }

    if (
      trimmedUser === ADMIN_USER.username &&
      trimmedPass === ADMIN_USER.password
    ) {
      setUsername(trimmedUser);
      setIsAuthenticated(true);
      localStorage.setItem(STORAGE_KEY_AUTH_USER, trimmedUser);
      localStorage.setItem(STORAGE_KEY_AUTH_STATUS, 'true');
      return { success: true };
    }

    const users = getStoredUsers();
    const match = users.find(
      (u) =>
        u.username.toLowerCase() === trimmedUser.toLowerCase() &&
        deobfuscate(u.password) === trimmedPass
    );

    if (match) {
      setUsername(match.username);
      setIsAuthenticated(true);
      localStorage.setItem(STORAGE_KEY_AUTH_USER, match.username);
      localStorage.setItem(STORAGE_KEY_AUTH_STATUS, 'true');
      return { success: true };
    }

    return { success: false, message: 'Incorrect username or password.' };
  };

  const logout = () => {
    setUsername('');
    setIsAuthenticated(false);
    localStorage.removeItem(STORAGE_KEY_AUTH_USER);
    localStorage.removeItem(STORAGE_KEY_AUTH_STATUS);
  };

  const value = { username, isAuthenticated, register, login, logout };

  return (
    <AuthContext.Provider value={value}>
      {children}
    </AuthContext.Provider>
  );
}

import { useNavigate } from 'react-router-dom';

function MovieCard({ movie, isSelected = false, onSelect }) {
  const navigate = useNavigate();

  if (!movie) return null;

  const handleCardClick = () => {
    navigate(`/movies/${movie.id}`);
  };

  const handleBookClick = (e) => {
    e.stopPropagation();
    if (onSelect) {
      onSelect(movie);
    }
  };

  const displayRating =
    movie.rating && movie.rating !== 'N/A'
      ? typeof movie.rating === 'number'
        ? movie.rating.toFixed(1)
        : movie.rating
      : null;

  return (
    <article
      className={`movie-card ${isSelected ? 'selected' : ''}`}
      onClick={handleCardClick}
      tabIndex={0}
      role="button"
      onKeyDown={(e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          handleCardClick();
        }
      }}
      aria-label={`View details for ${movie.title}`}
    >
      <div className="movie-poster-container">
        {movie.poster ? (
          <img
            src={movie.poster}
            alt={`${movie.title} poster`}
            className="movie-poster-img"
            loading="lazy"
            onError={(e) => {
              e.currentTarget.style.display = 'none';
              const fallback = e.currentTarget.nextElementSibling;
              if (fallback) fallback.style.display = 'flex';
            }}
          />
        ) : null}
        <div
          className="movie-poster-fallback"
          style={{ display: movie.poster ? 'none' : 'flex' }}
        >
          <span className="fallback-cinema-icon">🎬</span>
          <span className="fallback-title-abbr">{movie.title.slice(0, 18)}</span>

          <div className="movie-rating-badge" title={`Rating: ${displayRating} / 10`}>

          </div>
        )}

      </div>

      <div className="movie-card-info">
        <h3 className="movie-card-title" title={movie.title}>

        </h3>

        <div className="movie-card-genre">

import React from 'react';
import MovieCard from './MovieCard';
import Loading from './Loading';
import ErrorMessage from './ErrorMessage';
import EmptyState from './EmptyState';
import useBooking from '../hooks/useBooking';

function MovieList({
  movies = [],
  isLoading = false,
  error = null,
  onRetry,
  onReset,
  onSelectAndContinue,
}) {
  const { selectedMovie, handleSelectMovie } = useBooking();

  if (isLoading) {
    return <Loading type="skeleton" count={8} />;
  }

  if (error) {
    return (
      <ErrorMessage
        title="Failed to Load Movies"
        message={error}
        onRetry={onRetry}
        retryText="Retry Loading"
      />
    );
  }

  if (!movies || movies.length === 0) {
    return (
      <EmptyState
        title="No Movies Found"
        message="No cinema titles matched your current search query or genre filter. Try adjusting your search."
        onReset={onReset}
      />
    );
  }

  const handleMovieSelect = (movie) => {
    handleSelectMovie(movie);
    if (onSelectAndContinue) {
      onSelectAndContinue(movie);
    }
  };

  return (
    <section className="movie-list-section" aria-label="Movies Collection">
      <div className="movies-grid">
        {movies.map((movie) => {
          const isSelected = selectedMovie && String(selectedMovie.id) === String(movie.id);
          return (
            <MovieCard
              key={movie.id}
              movie={movie}
              isSelected={isSelected}
              onSelect={handleMovieSelect}
            />
          );
        })}
      </div>
    </section>
  );
}

import { useState, useEffect, useRef, useCallback, useMemo } from 'react';
import { useNavigate } from 'react-router-dom';
import useBooking from '../hooks/useBooking';
import SearchBar from '../components/SearchBar';
import CategoryFilter from '../components/CategoryFilter';
import GenreFilter from '../components/GenreFilter';
import MovieList from '../components/MovieList';
import { fetchMovies, getStandardGenres, getAvailableCategories } from '../services/movieApi';

function MoviesPage() {
  const navigate = useNavigate();
  const searchInputRef = useRef(null);
  const { selectedMovie, handleSelectMovie } = useBooking();

  const [searchQuery, setSearchQuery] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('All Movies');
  const [selectedGenre, setSelectedGenre] = useState('All Genres');
  const [movies, setMovies] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [isLiveApi, setIsLiveApi] = useState(false);
  const [page, setPage] = useState(1);
  const [totalPages, setTotalPages] = useState(1);
  const [loadingMore, setLoadingMore] = useState(false);

  const availableCategories = useMemo(() => getAvailableCategories(), []);
  const availableGenres = useMemo(() => getStandardGenres(), []);

  const loadMovies = useCallback(async (search, category, genre, pageNum = 1, signal) => {
    if (pageNum === 1) {
      setLoading(true);
    } else {
      setLoadingMore(true);
    }
    setError(null);

    try {
      const result = await fetchMovies({
        search,
        category,
        genre,
        page: pageNum,
        signal,
      });

      if (pageNum === 1) {
        setMovies(result.movies);
      } else {
        setMovies((prev) => [...prev, ...result.movies]);
      }
      setIsLiveApi(result.isLiveApi);
      setTotalPages(result.totalPages || 1);
    } catch (err) {
      if (err.name === 'AbortError') return;
      setError(err.message || 'Unable to fetch 2026 movies catalog.');
    } finally {
      setLoading(false);
      setLoadingMore(false);
    }
  }, []);

  const handleCategoryChange = (category) => {
    setSelectedCategory(category);
    setPage(1);
  };

  const handleGenreChange = (genre) => {
    setSelectedGenre(genre);
    setPage(1);
  };

  const handleSearchInputChange = (query) => {
    setSearchQuery(query);
    setPage(1);
  };

  useEffect(() => {
    const controller = new AbortController();

    const timer = setTimeout(() => {
      loadMovies(searchQuery, selectedCategory, selectedGenre, 1, controller.signal);
    }, 280);

    return () => {
      clearTimeout(timer);
      controller.abort();
    };
  }, [searchQuery, selectedCategory, selectedGenre, loadMovies]);

  const handleRetry = () => {
    loadMovies(searchQuery, selectedCategory, selectedGenre, page);
  };

  const handleLoadMore = () => {
    const nextPage = page + 1;
    setPage(nextPage);
    loadMovies(searchQuery, selectedCategory, selectedGenre, nextPage);
  };

  const handleResetFilters = () => {
    setSearchQuery('');
    setSelectedCategory('All Movies');
    setSelectedGenre('All Genres');
    setPage(1);
    if (searchInputRef.current) {
      searchInputRef.current.focus();
    }
  };

  const hasActiveFilters =
    Boolean(searchQuery.trim()) ||
    selectedCategory !== 'All Movies' ||
    selectedGenre !== 'All Genres';

  return (
    <div className="movies-page">
      <div className="page-header">
        <div className="page-header-row">
          <div>
            <div className="year-highlight-badge">
              <span>📅 2026 Cinema Season</span>
            </div>
            <h1 className="page-title">2026 Movies Catalog</h1>
            <p className="page-subtitle">
              Browse recent and upcoming 2026 Tamil and Hollywood blockbusters with TMDB integration
            </p>
          </div>

          <div className="api-status-badge" title="Data Source Mode">
            <span className={`status-dot ${isLiveApi ? 'live' : 'fallback'}`}></span>
            <span className="status-text">
              {isLiveApi ? 'Live TMDB API: Connected' : 'Mode: Local 2026 Demo Data'}
            </span>
          </div>
        </div>
      </div>

      <div className="movies-filter-panel">
        <SearchBar
          searchQuery={searchQuery}
          onSearchChange={handleSearchInputChange}
          searchInputRef={searchInputRef}
          placeholder="Search 2026 titles (e.g., Jailer 2, Doomsday, Mario Galaxy)..."
        />

        <CategoryFilter
          categories={availableCategories}
          selectedCategory={selectedCategory}
          onSelectCategory={handleCategoryChange}
        />

        <GenreFilter
          genres={availableGenres}
          selectedGenre={selectedGenre}
          onSelectGenre={handleGenreChange}
        />
      </div>

      <div className="results-meta-bar">
        <span className="results-count">
          {!loading && !error && (
            <>
              Showing <strong>{movies.length}</strong> {movies.length === 1 ? 'movie' : 'movies'}
              {selectedCategory !== 'All Movies' && ` in "${selectedCategory}"`}
              {selectedGenre !== 'All Genres' && ` • Genre: "${selectedGenre}"`}
              {searchQuery.trim() && ` • Matching: "${searchQuery.trim()}"`}
            </>
          )}
        </span>

        {hasActiveFilters && (
          <button
            type="button"
            className="clear-all-filters-btn"
            onClick={handleResetFilters}
          >
            Clear All Filters ✕
          </button>
        )}
      </div>

      <MovieList
        movies={movies}
        isLoading={loading}
        error={error}
        onRetry={handleRetry}
        onReset={handleResetFilters}
      />

      {!loading && !error && isLiveApi && page < totalPages && (
        <div className="load-more-container">
          <button
            type="button"
            className="btn-secondary load-more-btn"
            onClick={handleLoadMore}
            disabled={loadingMore}
          >
            {loadingMore ? 'Loading More Movies...' : 'Load More 2026 Movies ↓'}
          </button>
        </div>
      )}

      {selectedMovie && (
        <div className="continue-bar">
          <div className="continue-bar-info">
            <div className="continue-movie-icon">🎟️</div>
            <div>
              <div className="continue-bar-label">Selected for Booking</div>
              <div className="continue-bar-title">
                {selectedMovie.title}
                <span className="continue-bar-genre">
                  ({selectedMovie.category || selectedMovie.language} • {selectedMovie.genre} {selectedMovie.year ? `• ${selectedMovie.year}` : ''})
                </span>
              </div>
            </div>
          </div>

          <div className="continue-bar-actions">
            <button
              type="button"
              className="btn-secondary"
              onClick={() => handleSelectMovie(null)}
              title="Clear selection"
            >
              Clear Selection
            </button>
            <button
              type="button"
              className="btn-primary"
              onClick={() => navigate('/booking')}
            >
              <span>Continue to Seat Booking</span>
              <span>→</span>
            </button>
          </div>
        </div>
      )}
    </div>
  );
}

export default MoviesPage;

import { useState, useEffect } from 'react';
import { useParams, useNavigate, Link } from 'react-router-dom';
import { getMovieById } from '../services/movieApi';
import useBooking from '../hooks/useBooking';
import Loading from '../components/Loading';
import ErrorMessage from '../components/ErrorMessage';

function MovieDetailsPage() {
  const { id } = useParams();
  const navigate = useNavigate();
  const { selectedMovie, handleSelectMovie } = useBooking();

  const [movie, setMovie] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [retryTrigger, setRetryTrigger] = useState(0);

  useEffect(() => {
    let ignore = false;
    const controller = new AbortController();

    async function fetchDetails() {
      setLoading(true);
      setError(null);

      try {
        const data = await getMovieById(id, { signal: controller.signal });
        if (!ignore) {
          setMovie(data);
        }
      } catch (err) {
        if (err.name === 'AbortError') return;
        if (!ignore) {
          setError(err.message || 'Unable to retrieve movie details.');
        }
      } finally {
        if (!ignore) {
          setLoading(false);
        }
      }
    }

    fetchDetails();

    return () => {
      ignore = true;
      controller.abort();
    };
  }, [id, retryTrigger]);

  const handleBookNow = () => {
    if (movie) {
      handleSelectMovie(movie);
      navigate('/booking');
    }
  };

  const isCurrentSelection = selectedMovie && movie && String(selectedMovie.id) === String(movie.id);

  const genreList = Array.isArray(movie?.genres) && movie.genres.length > 0
    ? movie.genres.map((g) => (typeof g === 'object' && g ? g.name : String(g))).filter(Boolean)
    : typeof movie?.genre === 'string'
      ? movie.genre.split(/[/,•]/).map((s) => s.trim()).filter(Boolean)
      : [];

  const backdropUrl = movie?.backdrop || movie?.poster || '';

  return (
    <div className="movie-details-page">
      <div className="details-nav-bar">
        <button
          type="button"
          onClick={() => navigate('/movies')}
          className="btn-back-link"
        >
          ← Back to Movies
        </button>

        <div className="details-breadcrumbs">
          <Link to="/">Home</Link>
          <span className="crumb-sep">/</span>
          <Link to="/movies">Movies</Link>
          <span className="crumb-sep">/</span>
          <span className="crumb-current">{movie?.title || 'Details'}</span>
        </div>
      </div>

      {loading && <Loading type="details" />}

      {!loading && error && (
        <div className="details-error-wrapper">
          <ErrorMessage
            title="Movie Not Found"
            message={error}
            onRetry={() => setRetryTrigger((c) => c + 1)}
            retryText="Try Again"
          />
          <button
            type="button"
            className="btn-secondary"
            onClick={() => navigate('/movies')}
            style={{ marginTop: '16px' }}
          >
            ← Return to Movies Catalog
          </button>
        </div>
      )}

      {!loading && !error && movie && (
        <div className="movie-details-card">
          <div
            className="details-hero-banner"
            style={{
              backgroundImage: backdropUrl ? `url("${backdropUrl}")` : undefined,
              backgroundSize: 'cover',
              backgroundPosition: 'center 25%',
            }}
          >
            <div className="details-hero-overlay"></div>
          </div>

          <div className="details-main-content">
            <div className="details-poster-col">
              <div className="details-poster-wrap">
                {movie.poster ? (
                  <img
                    src={movie.poster}
                    alt={`${movie.title} poster`}
                    className="details-poster-img"
                    onError={(e) => {
                      e.currentTarget.style.display = 'none';
                      const fallback = e.currentTarget.nextElementSibling;
                      if (fallback) fallback.style.display = 'flex';
                    }}
                  />
                ) : null}
                <div
                  className="details-poster-fallback"
                  style={{ display: movie.poster ? 'none' : 'flex' }}
                >
                  🎬
                </div>
              </div>

              <div className="details-poster-actions">
                <button
                  type="button"

import React from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';

// Context Providers
import { AuthProvider } from './context/AuthContext';
import { BookingProvider } from './context/BookingContext';

// Components
import Header from './components/Header';
import Footer from './components/Footer';
import ProtectedRoute from './components/ProtectedRoute';

// Pages
import HomePage from './pages/HomePage';
import LoginPage from './pages/LoginPage';
import RegisterPage from './pages/RegisterPage';
import MoviesPage from './pages/MoviesPage';
import MovieDetailsPage from './pages/MovieDetailsPage';
import BookingPage from './pages/BookingPage';
import SummaryPage from './pages/SummaryPage';
import NotFound from './pages/NotFound';

import './App.css';

function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <BookingProvider>
          <div className="app-wrapper">
            <Header />
            <main className="main-content">
              <Routes>
                {/* Public Routes */}
                <Route path="/"         element={<HomePage />}    />
                <Route path="/login"    element={<LoginPage />}   />
                <Route path="/register" element={<RegisterPage />} />

                {/* Protected Routes (Authentication Required) */}
                <Route
                  path="/movies"
                  element={
                    <ProtectedRoute>
                      <MoviesPage />
                    </ProtectedRoute>
                  }
                />
                <Route
                  path="/movies/:id"
                  element={
                    <ProtectedRoute>
                      <MovieDetailsPage />
                    </ProtectedRoute>
                  }
                />
                <Route
                  path="/booking"
                  element={
                    <ProtectedRoute>
                      <BookingPage />
                    </ProtectedRoute>
                  }
                />
                <Route
                  path="/summary"
                  element={
                    <ProtectedRoute>
                      <SummaryPage />
                    </ProtectedRoute>
                  }
                />

                {/* 404 Route */}
                <Route path="*" element={<NotFound />} />
              </Routes>
            </main>
            <Footer />
          </div>
        </BookingProvider>
      </AuthProvider>
    </BrowserRouter>
  );
}

export default App;
